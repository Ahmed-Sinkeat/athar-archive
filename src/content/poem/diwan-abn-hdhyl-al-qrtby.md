## وكأن درعك أنشئت من مزنة

من البحر الكامل

وكأنّ درعَكَ أُنشئت من مزنَةٍ … فيكادُ أن يعشى بها المُستلئمُ
وكأنّهم مما تدانوا والتقوا … رفٌ فتحسَبُها تهمُّ وتهجمُ
وكأنّ جسمَك من وراء حجابها … دينٌ يشحُّ بهِ تقيُّ مُسلمُ

## كأن ليلى وفي إعادة أنجمه

من البحر البسيط

كأن ليلى وفي إعادة أنجُمهُ … لقا تأوَهت في ظلمانِه شابا
كأن ليلى شريكي في الهوى فإذا … فكَرتُ فكر والبلوى لمن خابا
كأنَّ ليلى وصبحي فيه محتجبٌ … غيرانُ سدَّ على معشوقتي بابا

## ولي الشيب بعد عزل الشباب

من البحر الخفيف

ولَيَ الشَيبُ بعد عزلِ الشّبابِ … كُل ما كانَ حُكمُهُ للغُرابِ
فكأنّ الشَّبابَ عاهدَ شَيبي … فَهوَ مُستخلَفٌ لهُ في التَّصابي

## كأن ليلى مما طال جانبه

من البحر البسيط

كأنّ ليلى مما طالَ جانبُه … أخافَ صَبحي حتّى ضلَّ أو هَربا
كانّ صُبحي يخشى أن يُؤنبَهُ … أهلُ الهوى فاختفى بالليلِ وانتَقَبا

## لما رأت شعري تنير لونه

من البحر الكامل

لمّا رأت شعري تنيّر لونُهُ … ورأتهُ مُحتَجَباَ وراء حجاب
قالت خنَبتَ فقلتُ ثَيبي إنمّا … لبسَ الحِدادَ على ذَهابِ شبابي

## وأرى بقية منرتي قد فرقت

من البحر الكامل

وأرى بقيّة منرتي قد فُرّقت … ليُرى بها ريشُ الغُرابِ غريبا
كالطير لما نادأتها هِجمةٌ … للصُقرِ فرّت في الجهاتِ هروبا
أو كافتراقِ السفر في ديمومةٍ … لم يخرجَوا من قَفرها تأوَيبا

## شكا في ظهره حدبه

من مجزوء

شكا في ظهره حَدبَه … فقلت دعوهُ يا كذَبَه
جرابٌ بين فخذَيهِ تعلّق … تعلق صَيتَ الجَلَبه
فألقاهُ على كتفي … هِ فهو عليهِ كالعَقَبه

## بمحلة خضراء أفرغ حليها

من البحر الكامل

بمحلّةٍ خضراءَ أفرغَ حليّها ال … ذهبيّ ضاغَةُ قَطرها المسكُوب
بَسَقت على شَرفِ البلادِ كأنّما … قامت إلى ما تحتها بخطيب
والروضُ قد ألفِ النّدى فكأنّهُ … عينُ توقّفَ دمعُها لرقيبِ
مُتخالفُ الألوانِ يجمعُ شَملَهُ … ريحانُ ريحً ضبا وريحُ جَنوبِ
فكأنمّا الصَفراءُ إِذ تُومي إلى ال … بيضاء صبٍّ جانحُ لحبيبِ

## أنظر إلى الشيهم كيف انزوى

من البحر السريع

أنظُر إلى الشّيهمِ كيف انزوى … كأنّهُ جَولقةٌ في التُّراب
كأنّما شاهَد حرباً ففي … أوصالهِ دسكرةُ من حراب

## كأن نؤادي في يدي خفقانه

من البحر الطويل

كأنّ نُؤادي في يدِي خَفَقانهِ … فريسةُ ليثٍ قد تلاشَت من النّهبِ
كأنّ سراباً في ضلوعي وجاحماً … فهذا حكى شوقي وهذا حكى قلبي

## وقفت على الغصن الجديد كأنما

من البحر الكامل

وقَفِت على الغُصنِ الجديدِ كأنّما … تلهُو بهِ في الغَيمِ أو يلهُو بِها
وتستّرت في سروةٍ مُلتفّةٍ … حِجبت عنِ الأبصار شخصَ رقيبها
فكأنّما ريحُ الجَنوبِ تغايرت … ألا تُرى إلاّ لوقتِ هُبوبها
باتَت تُغازُلها فلمّا أصبحت … بَرزت لنا كالشّمسِ تبل غُروبها

## مخزومة في ثبج شخت

من البحر السريع

مخزُومة في ثَبجٍ شَختِ … كأنّما استُقصِيَ بالنَّحتِ
كأنّما آخرها نقطةٌ … ساقطةٌ من قلمِ المُفتي
شدّت على الأرضِ على أربع … تُشبهُ شَعرَ الطفلِ في النبتِ
مكدودةٌ ليس لها راحةٌ … وتقطعُ الأيامَ بالصّمتِ

## كأن عيونهن عيون عين

من البحر الوافر

كأنّ عُيونهنّ عيونُ عِينٍ … فواترُ قد سَكِرنَ بغير راحِ
يموتُ العذلُ في أهلِ التَّصابي … بهِنّ فما لأهلِ العشقِ لاحِ

## وكأن البنود أجنته الطير

من البحر الخفيف

وكأنّ البنودَ أجنته الطي … رِ يُرَفرِفنَ إِذ حَوَتها القُيودُ
وكأنّ المُحمرةَ اللّونِ في الأٌف … قِ خدود يَزينُها التّوريدُ
وكأنَّ العقاب والريحَ إلفا … نِ فمن ذا وضلّ ومن ذي صدود

## كأنما الخيل أوله ضوارمها

من البحر البسيط

كأنّما الخيل أولُهُ ضوارمُها … أسدٌ وبينُهما صُلحٌ قد انعقدا
كأنّما قمم الفرسان قد تركت … فيها النعامُ تريكاً عمَّها عددا
كأنّها وسيوفُ الهندِ تترعُها … طيرٌ تجاوبُ طيراً صيتاً غرَدا

## ومدركات ولم تلب وليس لها

من البحر البسيط

ومدركاتٍ ولم تلب وليسَ لها … روحٌ وتنصفُ من باغِ وإن بَعُدا
في كُلّ واحِدةٍ صوتُ إِذا لَهَجَت … بهِ أصابت مُراداً في الذي مَرَدا
كأنّ أولادها حنَّ إِذا انبعثت … لم تبقِ لا والداَ حيًا ولا وَلدا

## وجاثمة بين أيدي الملوك

من البحر المتقارب

وجاثمةٍ بين أيدي الملُو … كِ ليست تقومُ ولا تقعُدُ
إِذا عطشت جاءها وردُها … ولَيست على منهلِ تُورَد
فإن أخذت ريها أرضعت … بَنيها بثدي هُو الإثمدُ

## وقفت على عباء والجزع بيننا

من البحر الطويل

وقفتُ على عباء والجزعُ بيننا … لأنضُر من نار على البُعدِ تُوقَدُ
تقوم بطُولِ الرمحِ إن هبّت الضبا … وعندَ سكون الريح تهدا فتقعدُ
فَشبّهتُها في الحالتين بقارىءِ … إذا اعترضتهُ سجدةُ ظلّ يَسجدُ

## ومؤلف الأوصال يختلف الصدى

من البحر الكامل

ومؤلفِ الأوصالِ يختلفُ الصَدى … فيهِ فتحسبُ صوتَهُ تَغريدا
رقّت معانيهِ برقَّةِ أربعٍ … صارت عليه قلائداً وعقودا
فكأنّ بُليل صائفٍ في صدرهِ … يصل الأغاني مُبدياً ومُعيداَ

## غنى وفوق جناحيه سقيط ندى

من البحر البسيط

غَنى وفوقَ جناحَيهِ سقيطُ ندى … والغَيمُ يُنجزُ للحَوذانِ ما وَعدا
يهفُو بهِ خُوط رَيحانٍ تُغازلَه … في الجوَّ ريحٌ فتلوي متنَهُ أودا
إذا استقلَّ ومش الأرضَ تَحسبُهُ … مُصلياً إِن تلقَى سجدةً سَجدا
لهُ ثلاثةُ ألوانٍ تخالُ بها … زُمرُّدا وعقيقاً جاورا بَردا

## ومضمومة في الخيزران كأنها

من البحر الطويل

ومضمومةٍ في الخيزرانِ كأنّها … على يده من شعرهِ الفاحمِ الجعدِ
تنيفُ عليهِ قدَّها فكأنّها … قضيبٌ تعالى عن قضيبِ على بُعدِ
وقد خُنِقَت بالتَّبرِ حتّى كأنّما … تهمُّ بأن تشكو لهُ ضائق العَقدِ

## محمد هل جوادك في الجياد

من البحر الوافر

محمدُ هل جوادُكَ في الجيَّادِ … إِذا حصَلت إلاّ كالفُؤادِ
كأنَّ ضلوعهُ ممّا تعرّت … قِسىِّ وُترت يومَ الجلادِ

## وناحلة صفراء من غير علة

من البحر الطويل

وناحلةٍ صفراء من غير علّةٍ … لها لمةُ حمراءُ ذاتُ تَوقُد
تلوحُ عليها صُفرةٌ عسجَديَّةٌ … بها تتحلى بُأساً كلّ مَشهَدِ
بكت لُؤلؤاً ينهلُّ من كُلّ مَدفعٍ … فعادَ عليها كالجُّمانِ المُنضَّد
تموتُ وتحيا تارةً بعد تارةٍ … فما تأمُلن من غيرها في تجدُّدِ

## والأرض عاطرة النواحي غضة

من البحر الكامل

والأرضُ عاطرةُ النّواحي غضّةٌ … خضراءُ في ثوب أغرَّ جديدِ
والماءُ تدفعُهُ إليكَ مثاعبٌ … شتّى من الميثاءِ والجُلمُودِ
صافٍ على صفةِ المها ومذاقُه … شَهدٌ فخُذ من طيِّبٍ وبَرودِ
ملأ التِّلاع فأقبلت وكانّها … هجماتُ حيّاتٍ ذوات حقُودِ
تنحو إلى حال الغطيطِ ورُبّما … زَأرت فتسمعها زئيرَ أسودِ
وَتُثيرُ طافيةَ الحصى فكأنّها … ولت على الساعاتِ فَهمَ بليد

## ويعيرك القلم المعلى واعيا

من البحر الكامل

ويُعيرك القلمُ المُعلّى واعياً … أذنَ المُحبِّ إلى الحبيبِ الأغيدِ
ليسَ السَّقامَ ولم يُكابد في الهوى … عشقاً ولم يشهد بوالي تَهمَدِ
وكأنّما كتمَ الهوى فاختالَ في … دمعٍ خلاف الدّمعِ داجٍ أسوَدِ

## روحني عاذلي فقلت له

من البحر المنسرح

روّحني عاذلي فقلتُ له … مه لا تزدني على الذي أجدُ
أما ترى النّارَ وَهيَ خامدةٌ … عند هبوب الرّياحِ تتقدُ

## كلفتها طول السهاد فراقبت

من البحر الكامل

كلّفتُها طولَ السُّهاد فراقَبت … برقاً يلوحُ وتارةً يتَستَّرُ
وكأن ليلى فارسٌ في كفهِ … رُمحٌ يقلّبُهُ عليهِ مِغفَرُ
تبدو لهُ شُعبٌ تطيرُ أمامها … شُغلٌ تطيرُ لها القلوبُ وتُذعرُ
ويروغُ عن قبض السّحابِ وميضُهُ … فكأنّهُ فرسٌ مُعارٌ أشقرُ

## ومحدوبة في كل وقت ظهورها

من البحر الطويل

ومحدوبةٍ في كُلّ وقتِ ظهورُها … نخافُ عوادي غدرها فنديرُها
لعزّت فلم يستَغنِ عنها ابنُ آدم … وهانت عليهِ فَهوَ لا يستعيرُها
كأنّ الذي يحتالُ في ردِّ روحها … مُناج لها أو صاحبٌ يَستشيرها
كأنّ ركاماً فوقها وَهيَ تحته … عجاجٌ وطِرفلإ اشقرّ يَستَثيرُها

## ومد ضبعيه في أعلى مزاحمة

من البحر البسيط

ومدَّ ضَبعَيهِ في أعلى مزاحمة … للنجم ما كانَ عنها النجمُ ينحدرُ
كأنّما هُوَ فيها شَخص مُستَرِقٍ … مُوقَّفٍ لبني الدنيا ليعتبروا

## وانظر إلى ملك النصارى كلها

من البحر الكامل

وانظر إلى ملكِ النّصارى كُلِّها … كيفَ استقادَ إليكَ وَهوَ مُبادرُ
والخوفُ يَقذفُهُ إليك كأنّهُ … كرةٌ لها في الصَّولجانِ مكاسرُ

## ومطرد الأعلام خال سرابه

من البحر الطويل

ومطّرِدِ الأعلامَ خال سرابُهُ … على الأمعزِ العاري برودٌ تُنَشَّرُ
كأنّ روابيهِ إذا اتُزرت بهِ … رجالٌ ببيضِ الرّيطِ ظهراً تأزّروا

## أكابد ليلا لا يزال كأنه

من البحر الطويل

أكابُد ليلاً لا يزالُ كأنّهُ … لإكبابهِ فوقي شجيٌ مُفكّرُ
واسألُهُ أن ينجلي فكأنّهُ … رثي لي ففيما نابني يتفكرُ

## سيقوا إليك فلو سقت قلوبهم

من البحر البسيط

سيقوا إليك فلو سُقت قلوبُهُمُ … لا سود ظنُّك من إفراط ما ستروا
يرومُ اخطبهُم تأليفَ واحدة … كأنّما بين فكّي نُطقِهِ حجرُ

## ترى قطرات الطل كالدر فوقها

من البحر الطويل

ترى قطراتِ الطلِّ كالدرِّ فوقها … إذا انتفضت في الأيكِ تنثرُهُ نثرا
إِذا فرقتهُ ألّفَ الغيم غيرَهُ … عليها فقد شبهتُها قينةُ سكرى
تزاحمُ أخرى مثلها بعقودها … ولم ترضَ باسترجاعِ منثورِها كِبرا

## كأني من فرط الصبابة عاشق

من البحر الطويل

كأنّي من فرطِ الصّبابةِ عاشقٌ … يخاف عليه كاشحاً فهو مُضمَرُ
إذا عادني من لستُ أنساهُ لم يجد … سوى أدمُع لم يدرِ من حيثُ تقطرُ
ويعلمُ أنّي قائمُ الشّخصِ كُلمّا … أحنُّ إلى ذكر الحبيبِ وأزفرُ
كما الرّيحُ إن هبّت سمعتَ هبوبها … وليس يراها ناظرُ حين تخطرُ

## وأغضف يلغي أنفه فكانما

من البحر الطويل

وأغضَفَ يلغي أنفَهُ فكانّما … يقُود به نورٌ من الوحي نَيَّرُ
إِذا ألهَبتُه شهوةُ الصَّيدِ طامعاً … رأيتَ عقيمَ الرّيحِ عنهُ تُقَصّرُ

## من كل ضافية الغدير ترى لها

من البحر الكامل

من كُلّ ضافيةِ الغدير ترى لها … طُرُقاً تصيرُ على المتُونِ غدائرا
قد سُمِّيت أُمّ الزمانِ فَأرضعت … أولادها ثديَ الرّماحِ أصاغرا
فكأنّهم يتطافرون لريبةٍ … أو رَوعةِ لو انّ فيهم طافراً
وكأنّهم مما تدانو والتقؤا … طيرٌ رأت في الجّو صقراً كاسرا

## كأن حناياها جناها مصفق

من البحر الطويل

كأنّ حناياها جناها مُصفّقٍ … إِذا الهبتهُ الشّمسُ أرخاهما نَشرا
كأنَّ سواريها شكت فترة الضنى … فباتَت هُضيماتِ الحشا نُحلاً صُفرا
كأنّ الذي زانَ البياض نحورَها … يُعذَّبها هجراً ويقطعها كبرا
كأنّ النخيلَ الباسقاتِ إلى العُلا … عذارى حجال رجّلت لمماً شُقرا
كأنّ غصونَ الآس والريح بينها … متونُ نَشاوى كُلّما اضطربت سُكرا
حديقةُ نفسٍ تملأ النفسَ بهجةً … وتُثني عيونُ الناظرين بها حشري
كأنّ جنى الجلّنار ووردَها … عشيقانِ لما استجمعها أظهرا خَفرا
كأنّ جنى سوسانها في سنا الضحى … كؤوس من البلّور قد حُشيت تبرا
كأنّ عيونَ النرجس الغضّ بالنّدى … عيونٌ تداري الدَّمعَ خيفة أن يُدري
كأنّ جنى الخيري في غبش الدُجى … نسيمُ حبيبٍ زار عاشقَهُ سرّا
كأنّ ينابيعَ المياهِ مراجلٌ … تفورُ وقد أذكت لهنّ الحصى جمرا

## يخالف العهد في تصرفه

من البحر المنسرح

يُخالفُ العهدَ في تصرُّفِهِ … وَهوَ على خَلقِهِ وإن صَغرا
كانّه في يَدَ بمحركهِ … ينشُرُ قَلبي بهِ وما شَعَرا
كأنّه داودَ حينَ يُوقِظُه … يقرأُ فيهِ الزّبورَ والسُّورا

## إن وهب بن مسره

من مجزوء

إن وَهب بن مسرّه … بين أهلِ العلمِ دُرّه
كانَ في مَجلِسه اليَو … مَ على العِلمِ معَرَّه
إذ علا القيّمُ رأ … سَ البثرَ ليِّ بدُرّه

## ويسل فيه من العنب الغض

من البحر الخفيف

ويسَلِّ فيهِ من العِنَبِ الغضِّ … شبيهُ العُنابِ في الاحمرارِ
رقٌ منهُ أديمهُ فَهوَ كاليا … قُوتِ يُستامُ بين أيدي التِّجارِ
وغَذَتهُ الأيامُ فَهوَ أنابي … ب طوال على جفانٍ قِصارِ

## من بنات الكروم ليس لها خمس

من البحر الخفيف

من بناتِ الكروم ليس لها خَم … سُ ليالٍ بكرٌ من الأبكارِ
يَتغنّى نَشيشُها في الرواقي … دِ فتُنسيكَ نغمةَ الأوتارِ
واستهلّت رفقاً كما يقعُ الطلُّ … على الورد في دُجى الأسحارِ
تَتَبدّى من خُبتها وهي صفراً … كبدوِّ الخيريِّ في الاصفرارِ
ثُم سَلسلتُها إلى جَسدٍ مَي … تٍ فأحيَتهُ فاعتبر باعتباري
باتَ بعد الخشُوعِ مُستِنِدَ الظَّه … رِ حطيئاً إلى أساسِ الجدارِ
ذوعكاكينَ رُكبّت كعكاكي … رَ حطيئاً إلى أساسِ الجدارِ
وشَدَدنا خِناقَهُ فَهوَ كالمِع … صَمَ ريّانُ في شدادِ السّوارِ

## ومرنة بعد الرواح كأنما

من البحر الكامل

ومُرِنَةٍ بعد الرّواحِ كأنّما … في نَحرِها صوتُ القريعِ الهادرِ
قرًبت من الأسماع وَهيَ بعيدةٌ … منها وغابت في الهبوب الحاضر
فإذا التقى جمهورها في دوحةٍ … فكأنَّ كُلّ ليثِ هاصرِ
وإِذا استقلّ قتامُها فكأنّما … فيهِ التفافُ عساكرٍ بمساكرٍ

## أساء إلى جفني فؤادي بناره

## وتشفق الدرع أن تنساب خائفة

## إذا نشرت كانت على دارة البدر

## يرنو وقد ألف الكرى فكأنما

من البحر الكامل

يرنو وقد ألِفَ الكرى فكأنّما … يَرنو إليك بمُقلتَي يَعفُورِ
وانسابَ في الخدِّ الأسيلِ عِذارُهُ … فكأنّه مسكٌ على كافُورِ

## أرى أهل الثراء إذا توفوا

من البحر الوافر

أرى أهلَ الثّراءِ إذا تُوفُوا … بَنَوا تلك المراصد بالصُّخُور
أبَوا إلاّ مُباهاةً وفخراً … على الفُقراءِ حتّى في القُبُورِ
فإن يكُنِ التّسامحِ في ذُراها … فإنّ العَدلَ فيها في القُعُورِ
عجبتُ لمن تأنّق في بناءٍ … أميناً مِن تَصاريفش الدُّهُورِ
ألَم يَبصُر بما قد خرَبتهُ ال … دُهوز من المدائن والقُصُورِ
وأقوامٍ مَضوا قَوماً مُقَوما … وصارَ صَغيرهُم إثرَ الكبيرِ
لَعَمرُ أبيهمُ لو أبصرُوهُم … لما عرفُوا الغنيَّ منَ الفقيرِ
ولا عَرَفوا العبيدَ مِنَ الموالي … ولا عَرفوا الإناث من الذكورِ
ولا مَن كانَ يلبسُ ثوبَ صوفٍ … مِنَ البَدنِ المُباشرِ للحريرِ
إذا أكلَ الثّرى هذا وهذا … فما فَضلُ الجليلِ على الحقير

## طيلساني طائر من نفسي

من البحر الرمل

طَيلَساني طائرٌ من نَفَسي … هو فَوقي غَبَشٌ في غَلَسِ
والذي ألَفهُ ألّفهُ … من هواءِ فارغٍ أو نَفَسِ

## أنا في الصيف راحة للنفوس

أنا في الصَّيفَ راحةٌ للنفوسِ … وشفاءٌ مَن حرٍّ داءٍ الرَّسيسِ
أنا زَينٌ في الكفِّ ساعةَ أُجلى … ليس مثلي يحلُّ كفِّ الرئيسِ
جلٌ شكلي عن أن أُنافس فيهِ … وجمالي يُزري بكُلِّ نفيسِ
غُرّتي البدرُ حين أبدو وجسمي … فضّةٌ رُكبّت على أَبَنوسِ

## وراقصة أسبلت لمة

من البحر المتقارب

وراقصةٍ أسبلت لِمّةً … عليها تُؤنّق في قصِّها
إذا حرّكتها لذبٍّ يدُ … تغنّى الذّباب على رقصها
فإن رُمتَ تُحصي خصالاً لها … لدى ذلك الرقصِ لم تُحصها

## وضعنا على جمر الفراق خدودنا

من البحر الطويل

وَضعنا على جمرِ الفراقِ خدودَنا … فعادت سماءُ الكِبر من ذُلنا أرضا
وَقفنا وقوفَ الدَّمع من بَهتةِ النّوى … فلم نستطع ركعاً ولم نستطع نَهضا

## خليلي ما انفك الأسى منذبينهم

من البحر الطويل

خَلِيليَّ ما انْفَكَّ الأَسَى مُنذُبينهِم … حَبِيبي حتَّى حَلَّ بالقَلْبِ فاخْتَطَّا
أُرِيدُ دُنُوّاً من خَلِيلي وقد نأَى … وأَهْوَى اقْتراباً مِن مَزارٍ وقد شَطَّا
وإِنِّي لَتَعْرُونِي الهُمُوم لذِكْرِكُم … هُدُوّاً فلا أَسطِيعُ قَبْضاً ولا بَسْطَا
وإِنَّ هُبُوطَ الوادِيَيْنِ إِلى النَّقَا … بحَيْثُ الْتَقَى الجَمْعَانِ واسْتَقْبَل السَّقْطَا
لِمَسْرَحِ سِرْبٍ ما تَقَرَّى نِعَاجُهُ … بَرِيراً ولا تَقْرُو جَآذِرُهُ خَمْطَا
ومُرْتجِزٍ أَلْقَى بِذِي الأَثْلِ كَلْكَلاً … وحَطَّ بجَرْعَاءِ الأَبَارِقِ ما حَطَّا
سَعَى في قِيادِ الرِّيحِ يَسْمَحُ للصَّبا … فأَلْقَتْ على غَيْرِ التِّلاعِ بِهِ مِرْطَا
وما زالَ يُرْوِي التُّرْبَ حَتَّى كَسا الرُّبى … دَرانِكَ والغِيطانَ مِن نَسْجِهِ بُسْطَا
وعَنَّتْ له رِيحٌ تُسَاقِطُ قَطْرَهُ … كما نَثَرَتْ حَسْنَاءُ مِن جيدهَا سِمْطَا
ولَم أَرَ دُرّاً بَدَّدَتْهُ يَدُ الصَّبا … سِواهُ فباتَ النَّوْرُ يَلْقُطُهُ لَقْطَا
وبِتْنَا نُرَاعِي اللَّيْلَ لم يَطْوِ برْدَهُ … ولَم يَجْرِ شَيْبُ الصُّبْحِ في فَرعِهِ وخْطَا
تَراهُ كَمَلْكِ الزَّنْجِ في فَرْطِ كِبْرِهِ … إِذا رامَ مَشْياً في تَبَخْتُرِهِ أَبْطَا
مُطِلاً على الآفَاقِ والبَدْرُ تاجُهُ … وقد عَلَّقَ الجَوْزاءَ في أُذْنِهِ قُرْطَا

## قصور إذا قامت ترى كل قائم

من البحر الطويل

قصورٌ إذا قامت ترى كُلّ قائم … على الأرضِ يَستخذي لها ثم يَخشعُ
كأنّ خطيباً مُشرفاً من سموكها … وشمٌ الرُّبى من تحتها تتسمعُ
ترى نورها من كُلّ بابٍ كأنّما … سنا الشمس من أبوابها يتقطعُ
ومن واقفاتٍ فوقهنّ أهلّةٌ … حنايا هي التّيجانُ أو هيَ أبدعُ
على عمدٍ يدعوكَ ماء صفائها … إليهِ فلولا جمدُها كُنتَ تكرعُ
تبوحُ بأسرار الحديثِ كأنّها … وشاةٌ بتنقيلِ الأحاديث تُولعُ
كأنّ الدكاكينَ التي اتصلت بها … صفائحُ كافورٍ تضيءُ وتَسطَعُ
كأنّ الأسودَ العامريةَ فَوقها … تهمُّ بمكروهٍ إليكَ فتفزعُ
كانَّ خريرَ الماء من لهواتها … تَبدّدُ دُرٌ ذابَ لو يتجمّعُ
أعدّت لإحياءِ البَساتين كلّما … سَقَت موضعاً منها تأكد موضعُ
دَعتها بصوبِ الماءِ فانتبهت لهُ … عُيونٌ كأمثال الدّنانيرِ تلمعُ
فلما نشا النّوار فيها ظَننتُها … قبابك يا منصورُ حين تُرفعُ
ولما اكتَست أغصانُها خِلتُ أنّها … قيانٌ بزيِّ أخضرٍ تتقنّعُ
ولما تناهى طِيبُها وتمايلت … علينا حسبناها حبيباً يُودَّعُ

## وسابغات كأنما نسجت

من البحر المنسرح

وسابغاتٍ كأنّما نُسِجت … بالآلِ ممَا صَفا مُلمّعها

## ويوما بدارات العقيق لو أنه

من البحر الطويل

وَيوماً بداراتِ العقيق لو أنّهُ … اُعيدَ لردِّ الشَّمس عن كُلّ مطلعِ

## متوسط جوز الغلاة كأنه

من البحر الكامل

مُتوسّطٌ جَوزَ الغلاةِ كأنّه … ثَملٌ يَميدُ بهِ الطريقُ المَهيَعُ
وترى بها جسم السّراب كأنّما … نَزلت به الحُمى فما إن تُقلِعُ

## ومرهفات كأنها شهب

## وبيني وبين المستقلة بالنوى

من البحر الطويل

وبَيني وبينَ المُستقلّةِ بالنّوى … دمُوعي ونورٌ ساطعٌ تحتَ بُرقُعِ

## لحق السها في جذعه فكأنه

من البحر الكامل

لحِقَ السُّها في جذعهِ فكأنّهُ … مُتسّمعٌ يَغشاهُ نجم فاذفُ

## ومحجل حر كأن أديمه

من البحر الكامل

ومُحجَّلٍ حُرٍّ كأنّ أديمهُ … سَبجٌ يكادُ يَسيلُ مما يلصف

فإذا هَفَت من فوقها تحكي لنا … قمراً يُغيبُ بالظلامِ ويكسفُ
ملآنُ من ريعانهِ فكأنّهُ … رَشأُ لأخفى نبأةٍ يتشوّفُ

## مروا كما محبوبي فمال تعج

من البحر الخفيف

مرُّوا كما محبُوبي فمالَ تَعُج … نحوي ركابُهُمُ ولم يتوقّفُوا

فعصيتُ إقدامي فما ودَّعتُهُ … إلاّ مُخالسة وعيري ترسُفُ

## وأنت ابتدعت لناعورتين

كأنّهما صاحبا غِلظةٍ … وبينهما عاشقٌ ملطفُ

فمدّت إلى أرضها ثديَها … مع السدِّ فَهوَ الذي يرشفُ
وبينهُما مجلسٌ للملوكِ … بهِ من عزازتهِ يُحلفُ
على قاعهِ لُجّةٌ من رخامِ … يُغرّقُ فيها ولا يتلفُ

ويُبطىءُ كالطّائر المستديرِ … عليهم ولكنّهُ يكنفُ

## لعبت بأيام الزمان وطاولت

## تعلقن بالأشفار من كل فعبرة

وأظنّهُ النّهرَ الذي لم يستطع … يحكيك في إرهامكَ الوكّافِ

## ويل كفكر في إقامة دولة

من البحر الطويل

وَيلٍ كفكرٍ في إقامةِ دُولةٍ … فلو كانَ في عِرقٍ لما نبضَ العِرقُ

## شاهدتهم وأنا أخاف عناقهم

ولَوِ أنّ عُذرةً شاهَدت من موقفي … شيئا لحذرها بأن لا تَعشَقا

## كأن الدروع البيض والبيض فوقها

من البحر الطويل

كأنّ الدّروعَ البِيضَ والبيضُ فَوقها … غَمائمُ غُرّ أفرَجَت عن بَوارقِ

## قد اختفى بين أغصان وأوراق

من البحر الكامل

## ومصروفة عن خلقها إن صرفتها

من البحر الطويل

ومَصرُوفةٍ عن خَلقِها إن صَرفتَها … إلى طيِّ بُرد أو إلى طيِّ مُهرَقِ

لها لُطفُ أنفاسِ الصّباحش ورِقّةٌ … تلذُّ بها نفسُ الفتى المُتشوِّقِ

## فكأنما فيه بقية روحه

من البحر الكامل

قد قابلَ الجهةَ التي كانت لهُ … فكأنّه باكٍ وإن لم يشهقَ

## ونأخذ منه جوده تحت هيبة

من البحر الطويل

ونأخذُ منهُ جودهُ تحتَ هيبةٍ … هيَ المُزنُ يَسقي الأرضَ والرّعدُ مُطبقُ

## صنعت كأجنحة الحمائم خفة

من البحر الكامل

وتكلّمت تحتَ القضيبَ كأنّما … نغماتها من حنةِ المُتشوقِ
يتكسرُّ الماشي بها فترى لهُ … خُيَلاءَ جبّارٍ وخِفّةَ أولقِ
ويُؤخّرُ الأقدامَ بعد تقدُّمِ … رَقص الحباب على الغديرِ المُتأقِ

## في نصفها من خجلها حمرة

من البحر السريع

## ومرنة والدجن ينسج فوقها

من البحر الكامل

ومُرنّةٍ والدَّجنُ ينسجُ فوقها … بُردين من طلٍ ونوءٍ باكِ

وترنمت لحنين قد حلتهما … بِغناءِ مُسمِعةٍ وأنّةِ شاكِ
فَفقدتُ من نَفسي لفرطِ تلهُّفي … نَفس الحياةِ وقلتُ من أبكاكِ

## من أين أقبلت يا من لا شبيه له

من البحر البسيط

من أينَ أقبلتَ يا من لا شَبيه لهُ … ومن هُوَ الشّمسُ والدُّنيا لهُ فَلكُ

## مطوقة يغدو الندى في جناحها

من البحر الطويل

مُطوقةُ يغدو النّدى في جَناحِها … لآلىءَ ليست من نظامِ ولا سَلَكِ
إذا انتقلت عن أيكها فكأنّما … قوادمُها أجفانُ والهةٍ تبكي

## فأول ما يبدو فخلق سبيكة

من البحر الطويل

فأوّلُ ما يبدو فَخلقُ سبيكةٍ … مُخلصةٍ بيضاء أتقنها السَّبكُ
بَنَت نَفسَها فوقَ الزُّمرّذِ واقفاً … فلاحت كمثلِ الدِّر ضمنهُ السِّلكُ
جنى سُوسَن لولا سنا بشراتهِ … لما زيّنَ الأفواهَ تغرّ ولا ضحكُ

## إن الذي ولى ففر بنفسه

من البحر الكامل

إنّ الذي ولّى ففرٌ بنفسه … للخَوفِ مكشوفٌ بلا سِريالِ

طارت بهِ وكأنّما أوصالُها … ولّت بعَينيها مِنَ الإعجالِ
ركبَ الشمالَ مولّياً ولقلبِهِ … خفقانُ هاذلةٍ بريحِ شمالِ

## في خضرة مفترة في غرة

من البحر الكامل

في خُضرةٍ مُفترّةٍ في غُرّةٍ … كالصُّبحِ كشفَ عنه ليلٌ أَليلُ
يمشي العِرضنَةَ فهو يحكي بالطلى … كيف الصُّدود عن الحبيب فيُبِلُ

## وجه أغر كأنه بدر الدجى

من البحر الكامل

وجهُ أغرٌ كأنّه بدرُ الدُّجى … فعليه من نور السُّعودِ كمالُ
تتراحمُ اللحظاتُ في إشراقهِ … فكانّه فوق العُيونِ هلالُ

## قلق الفرند مشطب فكانما

من البحر الكامل

قَلِقُ الفِرندِ مُشطّبٌ فكانّما … يَعلو ويَهبط في شَباهُ مَنهَلُ

## كأنه طبق الدنيا إذا انبسطت

من البحر البسيط

كأنّه طبّق الدنيا إذا انبسطت … فرسانه لمغارٍ يَومَ إرسال

إذا انثنى بقفُولٍ ماج من عِظمِ … دهراً كأنَّ ذويهِ غيرُ قُفّالِ

## تترى رؤوسهم عليك كأنها

من البحر الكامل

تَترى رؤوسهمُ عليكَ كأنّها … نُغرّ تَواضَت فوقَ روسِ تلال

فاستقبلتكَ كانّما عن توبةٍ … خضَعت لو ارتفعت إلى الأمهالِ

## تعاورتهم نبال عن معابلها

من البحر البسيط

تَعاورتهُم نبالٌ عن معابلها … كالنَّحلِ أو كشآبيب الحيا الزّجلِ
في كلّ واجدةٍ نعيّ تمدُّ بهِ … من رنّةِ الوترِ يحكي رنّة الشَّكلِ

## وحانية من غير رحمى على طفل

من البحر الطويل

وحانيةٍ من غير رُحمى على طفلِ … يَعيش بلا أكل ويبقى بلا رسلِ

لها رنّةٌ في إثرهِ بعد فَقدهِ … فتَحسبُها تبكي عليهِ من الثَّكلِ

## ومهتبل بالجو والأرض مسرع

من البحر الطويل

تجلى عليها مُقبلاً فكانّما … رماها بصَعقٍِ أو بنجمِ المُقاتلِ
كأن يديهِ فيهما قوسُ نادفٍ … فتُدني من الأوتارِ ريشَ الحواصلِ

## ودنت في هبوبها مشية النشوان

من البحر الخفيف

ودَنَت في هبُوبها مِشيَةَ النش … وانِ حَيرانَ بالمدامِ الشّمولِ

واختفت عن فواطِنِ الخَلقِ حتّى … شبّهوها ضآلةً بنحولِ

## فصلن ممحضا تجد بين فكي

من البحر الخفيف

فَصِلَن مُمحِضاً تجد بينَ فكَي … هِ لساناً بهِ يراضُ الكلامُ

## وثقيلة الأوصال تحسب أنها

من البحر الكامل

وثقيلةِ الأوصالٍِ تَحسَبُ أنّها … فلكٌ يَضيق بصبرها حيزومُها

## عرفت بعرف الريح أين تيمموا

من البحر الطويل

أبيتُ سميرَ الفرقدينِ كأنّما … وسادي قُتادٌ أو ضجيعيِّ أرقمُ
وأحورَ وَسنانِ الجفونِ كأنّه … قَضيبٌ من الرِّيحانِ لدنٌ مُنعّمُ
نظرتُ إلى أجفانه وإلى الهوى … فأيقنتُ أنّي منهُنّ أسلمُ
كما أنّ إبراهيمَ أوّلَ نظرةٍ … رأى في الدراري أنّهُ سمينٌ يَسقمُ

## وماذا عليهم لو أجابوا فسلموا

من البحر الطويل

## وتلك الأساطيل المسخرة التي

من البحر الطويل

## وسخية تعطيك أقصى جهدها

من البحر الكامل

## فتى بارد الأشعار يفظع لفظه

## تكاتف حتى الطير حوله

## وقائمة فى يدي قائم

إذا لاح في حيزومهِ فكأنّهُ … عليه نظامُ فوق جيدٍ ومعصَمِ
إذا مرَّ لم يدخُل مَمرّاً كأنّما … سَقَوهُ مُداماً بالكبير المفدَّمِ

## له لسانان من قرن إلى قدم

من البحر البسيط

لهُ لسانانِ من قرنٍ إلى قَدمِ … لا ينطُقانِ بغير السّحرِ والحِكَمِ
كأنّ أولهُ من حيةٍ سكنَت … إلى لِيانةِ كفِّ غَضّةِ العَنَمِ

## والثريا دنت من البدر حتى

من البحر الخفيف

والثّريا دَنت منَ البدرِ حتّى … خِلتَها دارعاً يُديرُ مجنّا

## قام على اليسرى خطيبا بها

من البحر السريع

قامَ على اليُسرى خطيباً بها … ينطقُ عن جُملةِ ألحانِ
كأنّما يفرقُ من فَزعةٍ … في أوّلِ من نقرهِ الواني
كأنّه في فعلهِ عاشقاً … روّعهُ العشقُ بهجرانِ
كانّما الأنقارُ في نحرهِ … ميازبٌ في طستِ عقيانِ

## ومما يقطع الحيزوم عندي

من البحر الوافر

ومما يقطَعُ الحيزومَ عندي … مرُوري بالشّواءِ على الخوانِ
وتندى بُردتي خَلفي إذا ما … نظرتُ إلى الهرائسِ في الجفانِ
كأنّ الزّيتَ والعسلَ المصفّى … عبيرٌ خالصٌ في دُهنِ بانِ
وباذنجان مثلِ كُراتِ ضَربٍ … تَضمّنها لُبابُ الجُلجُلانِ
وقد وقفَ الصّيام على فراغٍ … ونفسي سوفنَ تفرغُ بالأماني

## وكأن المقاتل اغتاظ حتى

من البحر الخفيف

وكأنّ المُقاتلَ اغتاظَ حتّى … أنفذَ الصُّبحَ بالتّقحّم طَعنا
والسُّهى في بناتِ نعشٍ ضميرٌ … بينَ أضلاعِها تَبّوأ كِنّا

## لا تلم هائما قد استحسن الوجد

من البحر الخفيف

لا تلُم هائماً قد استحسنَ الوج … دَ وكِل أمرَهُ إلى استحسانِه
فأنا الطائعُ المشوقُ منَ العُج … بِ به أن يُراعَ في ريعانة
مرّ بي خاطراً يكادُ من العُج … بِ به أن يُراعَ في ريعانه
في ملاءٍ كأنّهُ وَهوَ فيها … وَردُ خدّيهِ في جني سوسانِه
يتشكى الفتور من كسلِ المش … ي ولا يشتكيهِ من أجفانِه
ولقد شفّني وأسهر طَرفي … لمعُ برقٍ يَزفُّ في لمعانِه
شَمتُهُ والظلامُ يفترُ عنهُ … كافترارِ الزنجيّ عن أسنانِه

## قل لهذا الحمام إن جهل الحب

من البحر الخفيف

قُل لهذا الحمامِ إن جَهِلَ الحبِّ … أنا واقفٌ على عرفانِه
لم تُصبهُ النّوى بفقدانِ خلِّ … فيُرى باكياً على فقدانِه
فَشدا في قضيبِ أيكٍ يُعلَي … هِ ويُدنيهِ أرضَهُ من ليانِه
وكأنّ الرذاذَ فوق جناحي … هِ جمانٌ يروقُ عند اقترانِه

## وكأن الرايات وهي مع الريح

من البحر الخفيف

وكأنّ الرّاياتِ وهيَ معَ الرّي … حِ فؤادُ المقصُودِ في خَفَقانِه
وكأنّ الرذاذَ فوقَ جناحي … هِ جمانٌ يروقُ عند اقترانِه

## كأنا من الإجلال تحت عماية

من البحر الطويل

كأنّا من الإجلال تحتَ عمايةٍ … نُطاطي لها بالرُّعبِ كُلّ الأحاينِ
كأنّا قُرفنا باجترامٍ فما لنا … لسانٌ يُقوّينا بعذرٍ مُباينِ

## رب صغير الخلق ذي دهاء

من البحر الهزج

رُبَّ صغير الخلق ذي دهاء … يَستنزلُ الصَّير من السَّماء
داني المدى لغاية الثَّنائي … كأنّهُ ضربٌ من القضاءِ
إِذا هوى من خافقِ الهواءِ … سافٌ كمثلِ السَّيفِ في المضاءِ

## لما وضعت على قلبي يدي بيدي

من البحر البسيط

لمّا وضعت على قلبي يدي بيدي … وصحتُ في الليلةِ الظلماءِ واكبدي
ضجَت كواكبُ ليلي في مطالعها … وذابت الصخرةُ الصَماءُ من كمدي
وليس لي جُلَدُ في الحُبِّ ينصُرُني … فكيفَ أبقى بلا قلب بلا جلد
وكيف أشرحُ ما ذابَ الجَمادُ له … لمن غدا خائفاً إشارتي بيدي
لمّا رآني مُشيراً للسّلامِ بها … ألقى على خدّه مُضاعفُ الزّرَدِ

## على جيد الغزالة خلق جيدي

من البحر الوافر

على جيدِ الغزالةِ خلقُ جيدي … وأطرافُ الكواعبِ من عُقُودي
يَزيدُ الحنوَ في نفسي ونفسي … يُقالُ لها بحق الله زيدي
إذا هبّت أهازيجي صَبت لي … قلوبَ لسنَ من قلبِ العميدِ
وللأوتارِ في صدري حنينٌ … يهيجُ الشّوقَ في نفس العميدِ

## هبت لنا ريح الصبا فتعانقت

من البحر الكامل

هبت لنا ريحُ الصَّبا فتعانَقت … فذكرتُ جيدَكِ في العناقِ وجيدي
وإِذا تألّف في أعاليها النّدى … مالت بأعناقٍ ولطفِ قدودَ
وإِذا التقت بالرّيحِ لم تُبصِر بها … إلاّ خدوداً تلتقي بخدودِ
فكانّ عُذرةَ بينها تحكي لنا … صفة الخضوعِ وحالةَ المعمودِ
تيجانُها طلِّ وفي أعناقِها … منهُ نظامُ قلائدٍ وعقُودِ
فترشُني منهُ الصِّبا فكأنّهُ … من ماءِ وردٍ ليس للتَّصعيدِ
فكأنّما فيها لطيمةُ عاطرٍ … فتثيرُ ناراً في مجامرِ عودِ
شُغلت بها الأنداءُ حتّى خِلتُها … يَبسُطنَ أنديةً بها للصيدِ
وتجلبَبت زهراً فخِلتُ بأنّها … فوقي نثائرُ نادفٍ مجهُودِ
وتمتعت بذبابها فرياضُها … لبسِت كمثلِ المرتعِ المورودِ
غنّى فأسمعَني وغابَ فلم تَقع … عيني عليهِ في الكلا المنضُودِ
فكأنّ وترَ الموصليَّ ومَعبدٍ … بيديهِ فَهو يصوغ كُلَّ نشيدِ
يرقى إلى وَرَقِ الكلا وكأنّما … حيزومهُ من لمّةِ المولودِ
فكأنّهُ مشهّدٌ أو حاسبٌ … فَنِكٌ بعقدِ حسابهِ المكدودِ

## لا تلمني على البكاء بدار

من البحر الخفيف

لا تلُمني على البُكاء بدارِ … أهلها صيّروا السّقامَ ضَجيعي
جَعلوا لي إلى الوصالِ سبيلاً … ثم سدوّا عليّ بابَ الرجُوعِ

## يا فرجة للحادث المتكشف

من البحر الكامل

يا فرجةً للحادث المتكشفِ … ويداً يفيقُ بها الزمانُ وَيشتفي
عمَّ السرورُ فكلُّ نفسٍ حالُها … في حالِ يعقوبٍ ببُردةِ يُوسُفِ
لو كان شخصاً لم يعادل حسنهُ … حسن الربيع بزهرهِ المُتألفِ
ولو اللّيالي صورت أَيامها … منها لما اتَّصلت بداجٍ مُسرفِ
فرحٌ فما في العالمين مُولهُ … من حُسنِ موقعهِ اللّطيفِ المُلطفِ
قد لاحت الشمسُ التي أضواؤها … من عبد شمس في المحلِّ الأشرفِ
وقد استردّت رَيعَها ريحانةٌ … ذبلت فأيّة بلدةٍ لم تزحفِ
وقد استجبت دعوٌ لو أنّها … في العالمين لفضلها لم تخلفِ
قد كادَ يشمتُ بالشخاء وبالندى … بخلُ الأشحّةِ واللهُى بالمُعتفي
ضحِكت إلى تلكَ السّلامةِ وبالندى … تركت لعضد هِشامِها المُتخلَّفِ
قد كاد يَشمتُ بالمهنّدِ في الوغى … سردُ المعافر فوقَ كُلّ مُجفّفِ
يابن الخلائفِ من لُبابِ أميّةٍ … والمُستقّل بعزِّهِ والمكتفي
عُوفيتَ من كلّ الأذى ونعمتَ من … حُسنٍ وصرفُك وادعُ لم يُعنفِ

## وليس انبساطي في علاك مثقلا

من البحر الطويل

وليس انبساطي في عُلاكَ مُثقلاً … كغيري ولكن نيكَ جوهرُ منطِقي
فما أسالُ الحاجاتِ إلاّ كأنّما … حيائي على وجهي حُسامٌ بمفرقي

## في جانبي ليل وفي الثاني ضحى

من البحر الكامل

في جانبي ليلٌ وفي الثاني ضُحى … فأنا الزّمان على أناملِ مُمسكي
قرّب اليّ السيفَ لستُ أهابُهُ … وَدَعِ العيونَ فسيفُها هو مُهلكي

## ألا عودة من طيفه فيرى حالي

من البحر الطويل

ألا عودة من طيفهِ فيرى حالي … ألا يا أدّ كاري للكرى لو أتى تالي
يكادُ يضيقُ الجوُّ منعظم زَفرتي … وتهفُو نجومُ الليلِ من فرطِ إعوالي
أبى غيرَ تعذيبي ولو أمرض الرّدى … أطاعَ ولكن فعلهُ هُوَ أنكى لي

## لم يرحلوا إلا وفوق رحالهم

من البحر الطويل

لم يَرحلوا إلاّ وفوقَ رحالِهم … غيمٌ حكى غبشَ الصَّباح المُعتلي
وعلى هوادجهم مُجاجاتُ النّدى … فكأنّها مُطرت بدُرٌ مُرسلِ
لمّا تحرّكتِ الرّكابُ تناثرت … من فوقهم في الأرض بينَ الأرحلِ
فبكيتُ لو عرفوا دُموعي بينها … لكنّها اختلطت بشكلٍ مُشكِلِ

## وليل بغى فيه الغراب جناحه

من البحر الطويل

وليلٍ بغى فيهِ الغُرابُ جناحَه … ولم يَنفصل عنهُ ولكنّهُ عمي
دجاً فكأنّي من حناياهُ أو أتى … جريمةَ سوءٍ في سريرةِ مُجرمِ
إذا قُلتُ أَينَ الصُّبحُ فاضت سدولُهُ … عليَّ كانّي مُستغيثُ بأبكمِ
وأفزعُ مِن إطراقهِ فكأنّهُ … يُراصدُ إطلاقي نجيَّ التكتّمِ

## وماء كمثل الراح جار يزيدني

من البحر الطويل

وماءٍ كمثلِ الرّاح جارِ يزيدني … نشاطاً فيُجري كل معنى على ذهني
يمرُّ على حَبائه فكأنّهُ … صفا الدمعِ في عِقدِ الفتاةش التي أعني
