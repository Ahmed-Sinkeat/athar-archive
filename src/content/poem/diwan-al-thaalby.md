## يا حبذا حسنها ومرآها

من البحر المنسرح

يا حبَّذا حُسْنَها ومرآها … وحبَّذا في الثمارِ مَجْناها
تفَّاحةً في الكرى تُوافِقُني … وفي انتباهي فصِرْتُ أهواها
لأنَّها في المنامِ همَّةُ مَنْ … يأملُ مالاً ويبتغي جاها
وهيَ بهذي الأوصافِ ممتعةٌ … تريحُ رُوحي بطيبِ مرآها

## الليل أسهره فهمي راتب

من البحر الكامل

الليلُ أسهَرُهُ فَهَمِّي راتبُ … والصُّبْحُ أكرهُهُ ففيهِ نوائبُ
فكأنَّ ذاكَ قَذًى لطرفي مُسْهِرٌ … وكأنَّ هذا فيه سيفٌ قاضبُ

## ألا رب يوم بجرجان أرعن

من البحر الطويل

ألا رُبَّ يومٍ بجرجانَ أرعنٍ … ضحِكْتُ له من خَرْقِهِ أتعجَّبُ
وأخشى على نفسي اختلافَ هوائِهِ … وما للفتى ممَّا قضَى مهرَبُ
وما خيرُ يومٍ أخرقٍ متلوِّنٍ … ببردٍ وحرٍّ بعدَهُ يتلهَّبُ
فأوَّلُهُ للفَرْوِ والجمرِ مِثْقَبُ … وآخِرُهُ للثلجِ والخَيْشِ يُضْرَبُ

## ريق الحبيب كريق المزن والعنب

من البحر البسيط

ريقُ الحبيبِ كريقِ المُزْنِ والعِنَبِ … أَذَاقَني ثمراتِ اللَّهْوِ والطَّرَبِ
وقد سَبَتْ منِّيَ الأيَّامُ صفوتَها … فكيفَ أَهربُ منها وهيَ في طَلَبي

## قالوا تشوك خداه وشاربه

من البحر البسيط

قالوا تَشَوَّكَ خدَّاهُ وشاربُهُ … فقلتُ لا تُنْكِروا ما ليسَ بالعَجَبِ
الشَّوكُ في شجراتِ الوردِ مُحْتَمَلٌ … والشَّوكُ لا عَجَبٌ في مُجْتَنى الرُّطَبِ

## شيئان والله ما أملهما

من البحر المنسرح

شيئان واللهِ ما أمَلُّهما … وليسَ لي في سواهُما أَرَبُ
فإنْ تَقُلْ ما هُما أُجِبْ وأَقُلْ … بابُ خوارِزْمِشَاهَ والأَدَبُ

## وليل بته رهن اكتئاب

من البحر الوافر

وليلٍ بتُّه رهنَ اكتئابِ … أقاسي فيه أنواعُ العذابِ
إذا شرِبَ البعوضُ دمي وغنَّى … فللبرغوثِ رقصٌ في ثيابي

## وقصر ملك ترى كل الجمال به

من البحر البسيط

وقصرِ مُلْكٍ ترى كلَّ الجمالِ بهِ … وأَسْعُدُ الدَّهْرِ تبدو من جوانِبِه
كأنَّما جنَّةُ الفِرْدَوْسِ قد نَزَلَتْ … إلى خَوَارِزْمَ تعجيلاً لصاحِبِه

## لقد طرب المحب إلى الحبيب

من البحر الوافر

لقد طَرِبَ المحبُّ إلى الحبيبِ … كما طَرِبَ المريضُ إلى الطبيبِ
وأذعنَ للهوى القلبُ المعنَّى … كإذعانِ الشبيبةِ للمشيبِ
ألا يا حبَّذا مَسْرى نسيمٍ … نسيبِ الروحِ يا لكَ من نسيبِ
كأنَّ الروحَ منهُ طيبُ ذكرِ ال … عميدِ السيِّدِ الفردِ الأديبِ
أبي نصرِ بن مشكان الذي قد … غدا بينَ الأنامِ بلا ضريبِ
أتمَّ اللهُ نعمتَهُ عليهِ … ولا أخلاهُ من طربٍ وطيبِ

## فديتك يا أتم الناس حسنا

من البحر الوافر

فَدَيْتُكَ يا أَتمَّ الناسِ حُسْناً … وأصلَحَهم لمتَّخِذٍ حبيبا
فوجهُكَ نزهةُ الأبصارِ حُسْناً … وصوتُكَ مُتْعَةُ الأسماعِ طيبا
وسائلةٍ تسائِلُ عنكَ قُلْنا … لها في وصفِكَ العَجَبَ العَجيبا
دَنا ظبياً وغَنَّى عندَلِيبا … ولاحَ شقائِقاً ومشى قَضِيبا

## أقول والقلب مني في تلفته

من البحر البسيط

أقولُ والقلبُ منِّي في تلفُّتِهِ … يا بدرُ يا غائباً في أُفقِ مغربِهِ
نَذَرتُ لله صوماً إن رجعتُ وما … كفَّارةُ النَّذْرِ إلا بالوفاءِ بهِ

## لما بعثت فلم تسعف مطالبتي

من البحر البسيط

لما بعثتُ فلم تُسْعِفْ مطالَبتي … وأمْعَنَتْ نارُ شوقي في تلهُّبِها
ولم أجِدْ حيلةً تُبْقي على رَمَقي … قبَّلتُ عينَ رسولي إذ رآك بها

## يا ليلة طالت كأن نجومها

من البحر الكامل

يا ليلةً طالتْ كأنَّ نجومَها … غُرَماءُ أرقبَهُم لِدَيْنٍ واجبِ
والبَدْرُ كالشَّيْخِ الأجَلِّ تَمَنْطَقَتْ … قُدَّامَهُ الجَوْزاءُ مثلَ الحاجِبِ

## يا دهر حسبك قد أطلت نحيبي

من البحر الكامل

يا دهرُ حسبُكَ قد أطَلْتَ نحيبي … وَتَرَكْتَني في موطِني كغريبِ
وسَلَبْتَني ثوبَ السُّرورِ بجامعٍ … ما بينَ وَصْفَيْ خادمٍ وأديبِ
فالشِّعْرُ منِّي والدموعُ لآلئٌ … من نظمِ طَبْعَيْ عاشِقٍ وأديبِ
قد غابَ عن رَبْعي هلالٌ مُقْمِرٌ … في أُفْقِ تربِيَتي وفي تأدِيبي
فالآنَ يطلعُ في سِوى دارِي ولا … ينفكُّ فيهِ القلبُ رَهْنَ نحيبِ
نَدٌّ نفيسٌ عندَ غيري فائحٌ … وأراهُ من عَجْني ومن تركِيبي
وثمينُ عِقْدٍ عند غيري لائحٌ … وأراهُ من نَظْمي ومن تَرْتيبي

## من كان ينفعه الأدب

من مجزوء

من كان ينفعُهُ الأدَبْ … ويُحِلُّهُ أعلى الرُّتَبْ
فلقد خَسِرْتُ عليهِ ما … وُرِّثْتُ من أُمٍّ وأَبْ
كم ضيعةٍ كانتْ تصو … نُ الوَجْهَ عن ذُلِّ الطَّلَبْ
أتلَفْتُها لا في القِيا … نِ ولا هوَى بِنْتِ العِنَبْ
بل في الحوادِثِ والحوا … ئجِ والنوائبِ والنُّوَبْ
كم قُلْتُ لمَّا بِعْتُها … وحَصَلْتُ في أَسْرِ الكَرَبْ
ذَهَبَتْ دجاجَتُنا التي … كانَتْ تبيضُ لنا الذَّهَبْ

## كتبت وشينات حالي غلبن

من البحر المتقارب

كتبتُ وشيناتُ حالي غَلَبْن … عليَّ بمَنْ جَلَّ عن مُشْبِهِ
فشوقي إليهِ وشُكْري له … وشعريَ فيهِ وشُغْلي بهِ

## غناؤك يهزم جيش الكروب

من البحر المتقارب

غناؤكَ يهزِمُ جيشَ الكُرُوبْ … وعيناكَ للناسِ عذرُ الذُّنوبْ
فويلَ القلوبِ إذا ما رَنَوْتَ … وإمَّا شَدَوْتَ فويلَ الجُيوبْ

## وشادن أصبح عذر الذنوب

من البحر السريع

وشادنٍ أصبحَ عُذْرَ الذُّنُوبْ … لقاؤُه يهزِمُ جيشَ الكروبْ
بغُرَّةٍ غرَّارَةٍ للوَرَى … وطُرَّةٍ طَرَّارَةٍ للقُلُوبْ

## لي سيد ملك غدا

من مجزوء

لي سيِّدٌ مَلِكٌ غَدا … في بردَتَيْ مَلِكٍ وَهُوبِ
لا بالجهولِ ولا المَلُو … لِ ولا القَطُوبِ ولا الغَضُوبِ
قد جادَ لي بأَغَرَّ أُن … عِلَ بالشَّمالِ وبالجَنوبِ
لا بالشَّمُوسِ ولا العمو … صِ ولا القَطوفِ ولا الشبوبِ

## ولما نزلنا البشنقان التي غدت

من البحر الطويل

ولمَّا نزلنا البُشنقانَ التي غَدَتْ … وراحت بجنَّاتِ النعيمِ تشبَّهُ
وقد بَرَزَتْ أشجارها في ملابسٍ … ربيعيَّةٍ حازت مدى الحُسْنِ كلَّهُ
وعارَضَنا ماءٌ يروقُ مُصَنْدَلٌ … وواجَهَنا وردٌ يشوقُ مُوَجَّهُ
وقَهْقَهَ رَعْدٌ في السماءِ مُجَلْجلٌ … وفي الأرضِ إبريقُ المُدامِ يقهقِهُ
وغنَّى مغنِّي العندليبِ كأنَّما … يجاوِبُهُ في حلقِهِ مِزْهَرٌ لهُ
تنزَّهَ سمعي ما أرادَ وناظري … وقلبي من الأحزانِ لا يتنزَّهُ

## يا دار ملك نرى كل الجمال بها

من البحر البسيط

يا دار ملكٍ نرى كل الجمال بها … وأسعد الدهر تبدو من جوانبها
كأنما جنة الفردوس اذ نزلت … بأرض غزنة تعجيلا لصاحبها

## لقد قلب الدهر الخئون مجنه

من البحر الطويل

لقد قلب الدهر الخئون مجنه … فقلبي على جمر الغضى يتقلب
وأصبحت في ظفر الزمان ونابه … وما فيه الا دون ما أترقب

## وأرى الهلال بن الثلاث مطرزا

من البحر الكامل

وأرى الهلال بن الثلاث مطرزا … ثوب الدجى والجو في زرق العصب
فكأنما فرش الأمير المرتجى … ألقى بروض بنفسج نعل الدهب

## وكاتب كتبه تذكرني

من البحر المنسرح

وكاتب كتبه تذكرني ال … قران حتى أظلَّ في عجب
فاللفظ قالوا قلوبنا غلُف … والخط تبت يدا أبي لهب

## ألم تر أن الله أوحى لمريم

من البحر الطويل

ألم تر أن الله أوحى لمريم … وهزي إليك الجذع تساقط الرطب
ولو شاء أن تجنيه من غير هزها … جنته ولكن كل شيء له سبب

## وكنت أبكي قرير العين من فرح

من البحر البسيط

وكنت أبكي قرير العين من فرح … والآن من عجب في ضحك مكروب
وكنت أولع بالتصفيق من طرب … فالآن أوهى يدي تصفيق محروب

## لما رأيت زمانا

من البحر المجتث

لما رأيت زماناً … يفتر عن كل صعبه
والقحط في أكله النا … سَ بالذئاب تشبَّه
والحَبُّ قد عزَّ حتى … أنسى المحبَّ الاحبه
في حَبةِ القلب منّى … زرعت حُبَّ ابن حبَه

## سقيا لأيام الصبا إذ أنا

من البحر السريع

سقياً لأيَّام الصَّبا إذ أنا … في طَلَبِ اللَّذَّةِ عفريتُ
أصيدُ كالبازي ولكنَّني … أَسْفِدُ كالعُصفورِ ما شيتُ

## من ذا الذي لا يذل الدهر صعبته

من البحر البسيط

من ذا الذي لا يُذِلُّ الدهرُ صعبَتَهُ … ولا تُلينُ يدُ الأيامِ صَعْدَتَهُ
أما ترى خَلَفاً شيخَ الملوكِ غَدا … مملوكَ من فَتَح العذراءَ بلدتَهُ
قد كانَ بالأمسِ مَلكاً لا نظيرَ لهُ … فاليومَ في الأسْرِ لا ينتاشُ أُسرتَهُ

## كم حيلة للوصل أعملتها

من البحر السريع

كم حيلةٍ للوصلِ أَعْمَلْتُها … وكم خداعٍ قد تَمَحَّلْتُهُ
أُسِرُّ حَسْواً في ارتغاءٍ إذا … ناجَيْتُ مَنْ أهوى فقبَّلْتُهُ

## برأس سكة عمار لنا قمر

من البحر البسيط

برأسِ سكةِ عمَّارٍ لنا قَمَرٌ … من وجهِ عثمانَ يا طُوبى لجيرتِهِ
إذْ قوتُ أجسامِهِمْ ممَّا يبيعهُمُ … وقوتُ أرواحهِمْ من حُسْنِ صورَتِهِ

## الأرض طاؤوسية

من مجزوء

الأرضُ طاؤوسيَّةٌ … والجوُّ جُؤْجُؤُ فاخِتِ
متبسِّمٌ عن نَشْر حُ … بٍّ عندَ صَبٍّ ثابِتِ
والوردُ دُرٌّ نابتٌ … أحسِنْ بدُرٍّ نابِتِ
لكنَّ في عيني قذًى … من نورِ شيبٍ سابتِ
لما بكيتُ دَمَ الفؤا … دِ على الحبيبِ الفائتِ
ضَحِكَ المشيبُ بعارضي … ضحكَ الغَوِيِّ الشامتِ

## قولوا لعثمان في أوقات طيبته

من البحر البسيط

قولوا لعثمانَ في أوقاتِ طيبتِهِ … إذا تبسَّمَ عن دُرٍّ وياقوتِ
إنِّي أراكَ تبيعُ الناسَ قوتَهُمُ … ففيمَ تمنعُ عنِّي القوتَ يا قوتي

## كأنما النارنج للربات

من البحر الرجز

كأنَّما النارَنْجُ للربَّاتِ … ثَدِيُّ أبكارٍ مُخَدَّراتِ
مُزَعْفَراتٍ ومُعَصْفَراتٍ … أو أُكَرُ الكيمَخْتِ مذْهَباتِ
قد ضُمِّخَتْ بالعَنْبَرِ الفُتاتِ … نسيمُها يَزيدُ في الحَيَاةِ

## كم إلى كم تبرمي بحياتي

من البحر الخفيف

كمْ إلى كم تبرُّمي بحياتي … أتلوَّى تلوِّيَ الحيَّاتِ
تحتَ عبءٍ من الزمانِ ثقيلٍ … وخطوبٍ قوَّسْنَ منِّي قناتي

## قد لبس الدهر حسن زهوته

من البحر المنسرح

قد لبسَ الدهرُ حسنَ زهوتِهِ … مذ زُوِّجَ المُشتري بزهرَتِهِ
وفي اقترانِ العينِ ما فيه من … إشراقِ وجهِ العُلى ونضْرَتِهِ
فالطَّرفُ مستأنِسٌ بقُرَّتِهِ … والقلبُ يُطْوى على مَسَرَّتِهِ

## صفحت لدهري عن جميع هناته

من البحر الطويل

صفحت لدهري عن جميع هناته … وعددت يوم الباغ اسنى هباته
وقابلت أشجاراً هناك تقدمن … تعطل غصن البان عن حركاته
ويخجل ورد الباغ عن طلوعه … ويعذله بالورد في وجناته
ويسجد نور الاقحوان لثغره … ويقصر نشر الورد عن نفحاته
ولما دجى الليل استعاد سنا الضحى … بوجه جميع الحسن بعض صفاته
فيا لك من ليل رقيق ظلامه … بتأليف شمل الانس بعد شتاته

## لقاؤك يحكي قضاء الحوائج

من البحر المتقارب

لقاؤكَ يحكي قضاءَ الحوائِجْ … ووجهُكَ للهمِّ والغمِّ فارِجْ
وفيكَ لنا فِتَنٌ أربعٌ … تَسُلُّ علينا سيوفَ الخوارِجْ
لحاظُ الظباءِ وطوقُ الحمام … ومشيُ القباجِ وزيُّ التدارجْ

## ومدام قد كفانا

من مجزوء

ومدامٍ قد كَفَانا … شُغْلَ إشعالِ المسارِجْ
لو دنتْ منَّا القُماري … لاكتستْ ريشَ التدارُجْ
فاشربَنْهُ فهو للغُمَّ … ةِ والغَمَّاءِ فارِجْ
وهو ريقٌ من فم الدُّن … يا إلى ثغرِكَ خارِجْ

## يا كاتبا اقبل من زرنج

من البحر الرجز

يا كاتباً اقبل من زرنج … مبرقع الوجه بلون الزنج
اذهب فأنت بلغة الشطرنج

## أما ترى الدهر وأيامه

من البحر السريع

أما ترى الدهرَ وأيَّامَهُ … في العمرِ مثلَ النارِ في الشيحِ
يمرُّ كالريحِ وما في يدي … من مَرِّهِ شيءٌ سوى الريحِ

## تفاح غزنة نفاع ونفاح

من البحر البسيط

تفاحُ غزنةَ نفَّاعٌ ونفَّاحٌ … كأنَّهُ الشهدُ والريحانُ والراحُ
وماؤُهُ بادِّكارِ الرِّيقِ من قمرٍ … في خدِّهِ أبداً وردٌ وتُفَّاحُ

## ألا إن الغنى للمرء روح

من البحر الوافر

ألا إنَّ الغِنى للمرءِ روحُ … وإن غناءهُ في الأُذنِ ريحُ
وما بمُحَصِّلٍ عقلاً ودينا … ليذهبَ منهُ بينَ الريحِ روحُ

## يا خاتم الملك ويا قاهر

من البحر الطويل

يا خاتمَ الملكِ ويا قاهرَ ال … أملاكِ بينَ الأخذِ والصفْحِ
عليكَ عينُ اللهِ من فاتحٍ … للأرضِ مُسْتَوْلٍ على النُّجْحِ
راياتُهُ تنطِقُ بالنصرِ بل … تكادُ تُملي كُتُبَ الفَتْحِ
كَمْ أثرٍ في الدين أثَّرْتَهُ … يقصِرُ عنهُ أثرُ الصُّبْحِ
وكم عُلاً للمجدِ شيَّدْتها … تثني عليها ألسن المدْحِ
فاسعد بأيَّامك واستغرق ال … أعداءَ بالكبْحِ وبالذبْحِ
ودُمْ رفيعاً عاليَ القِدْحِ … ممتنعِ الملك عن القَدْحِ

## يا طبيبا منجما وفقيها

من البحر الخفيف

يا طبيباً منجما وفقيها … شاعراً شعره غذاء الروح
أنت طورا كمثل جامع سفيا … ن وطورا تحكي سفينة نوح

## أخوك هلال العيد عادت سعوده

من البحر الطويل

أخوكَ هلالُ العيدِ عادَتْ سعودُهُ … يحاكيكَ منهُ نورُهُ وصُعودُهُ
فأفْطِرْ على دهرِ بعينك ناظرٍ … وأبشِرْ بعيدٍ مورقٍ لك عودُهُ
وعيَّدتَ يا مَنْ للمعالي قيامُهُ … وللفضلِ والإفضالِ فينا قعودُهُ
يأيمنِ إهلالٍ وأسْعَدِ طالعٍ … وأكْمَلِ إقبالٍ يليهِ خلودُهُ

## دع الأساطير والأنباء ناحية

من البحر البسيط

دعِ الأساطيرَ والأنباءَ ناحيةً … وعاينِ الملكَ المنصورَ مَسْعُودا
تَرَ الأكابرَ طُرًّا والملوكَ معاً … ورُسْتُماً وسليمانَ بنَ داودا

## في روضة أزهارها مكسوبة

من البحر الكامل

في روضةٍ أزهارُها مكسوبَةٌ … والظلُّ من أشجارِها ممدودُ
فيها طرائفُ نرجسٍ وشقائقٌ … فكأنَّها من أعينٍ وخدودِ

## لله برد خوارزم إذا كلبت

من البحر البسيط

للهِ بردُ خوارِزْمٍ إذا كَلِبَتْ … أنيابُهُ وكسَتْ أبدانَنَا الرَّعدَا
فالشمسُ محجوبَةٌ والريحُ مُدْميةٌ … جلودَ قومٍ أضاعُوا الصَّبْرَ والجَلَدا
والماءُ مُسْتَحْجِرٌ والكلبُ منجَحِرٌ … والزمهريرُ يسوقُ الصِّرَّ والصَّرَدا
فلو تقبل معشوقاً مخالسةً … رأيتَ فاكَ على فيهِ وقد جَمَدا

## يا سيدا بالمكرمات ارتدى

من البحر السريع

يا سيِّداً بالمكرماتِ ارتدى … وانْتَعَلَ العَيُّوقَ والفَرْقدا
ما لكَ لا تَجري على مُقْتَضَى … مودَّةٍ طالَ عليها المَدَى
إن غِبْتَ لم أُطْلبْ وهذا سلي … مانُ بنُ داودٍ نبيُّ الهُدى
تفقَّد الطيرَ على شُغْلِهِ … فقالَ ما لي لا أرَى الهُدْهُدا

## أطال الإله بقاء الأمير

من البحر المتقارب

أطالَ الإلهُ بقاءَ الأمير … وتوفيقَهُ ثمَّ تأييدَهُ
ففي كلِّ يومٍ بإقبالِهِ … يرى عبدُهُ عندَهُ عيدَهُ

## وكلام كدمع صب غريب

من البحر الخفيف

وكلامٍ كدَمْعٍ صَبٍّ غريبٍ … رَقَّ حتى الهواءُ يكثُفُ عندَهُ
رقَّ لفظاً ودَقَّ معنًى فأضْحى … كلُّ سرِّ من البلاغةِ عندَهُ

## أبا منصور المغرور أقصر

من البحر الوافر

أبا منصورٍ المغرورَ أقْصِرْ … وأبْصِرْ طُرْقَ أصحابِ الرَّشادِ
ألَسْتَ ترى نجومَ الشيبِ لاحَتْ … وشيبُ المرءِ عنوانُ الفَسَادِ

## بنفسي هلال يخال الهلال

من البحر المتقارب

بنفسي هلالٌ يُخالُ الهِلالُ … لتلكَ المحاسنِ منهُ حَسُودا
كأنَّ عقاربَ أصداغِهِ … غُذِينَ بمسكٍ فأصبحنَ سُودا

## وأبلغ من عبد الحميد وجعفر

من البحر الطويل

وأبلغُ من عبدِ الحميدِ وجعفرٍ … ويحيى وإسماعيلَ أعني ابنَ عبادِ
فلا زالَ محروساً ولا زالَ ذكرُهُ … وأخبارُهُ أذكى من النَّدِّ في النادِي

## تم الكتاب بدولة الشيخ الذي

من البحر الكامل

تمَّ الكتابُ بدولةِ الشيخِ الذي … قد صُكَّ تاجُ علاه فوقَ الفَرْقَدِ
بدرِ الصدورِ مسافرٍ ركنٍ العلا … والمكرماتِ وكِيمياءِ السؤدَدِ
والحمدُ للهِ العظيمِ جلالُهُ … ثمَّ الصَّلاةُ على النبيِّ محمدِ

## أرسلت في وصف صديق لنا

من البحر السريع

أرسلتُ في وصفِ صديقٍ لنا … ما حقُّهُ الكتبة بالعسجدِ
في الحسنِ طاووسٌ ولكنَّهُ … أسْجَدُ في الخلوةِ من هُدْهُدِ

## برق مدام في عارض الند

من البحر المنسرح

برقُ مدامٍ في عارضِ النَّدِّ … ورعدُ قصفٍ وقَطْرُ ماءِ وردِ
والشمسُ مَعْ كلِّ هذِه طَلَعَتْ … من جيبِ ساقٍ مُمَشَّقِ القَدِّ
في فلكٍ دارَ قُطْبُ مركزِهِ … على نجومِ السرورِ والسَّعْدِ
لو تَمَّ أنسي بالقربِ منكَ لما … حَصَلْتُ إلاّ في جنَّةِ الخُلْدِ

## صديق لنا مذ ذقت طعم اخائه

من البحر الطويل

صديق لنا مذ ذقت طعم اخائه … غصصت وقد أربى على المر شهده
فأضعف من نسج العناكب عهده … وأضيع من نار الحباحب وده

## إذا قيل من فرد العلى والمحامد

من البحر الطويل

إذا قيل من فرد العلى والمحامد … أجاب لسان الدهر ذاك ابن حامد
همام له في مرتقى الدهر مصعد … يلوح له العيوق في ثوب حاسد
كريم حباه المشتري بسعوده … وأصبح في الآداب بكر عطارد
به سحبت خوارزم ذيل مفاخر … على خطة الشعرى وربع الفراقد
فلا زال في ظل السعادة ناعماً … يحوز جميع الفضل في شخص واحد

## إني أحاجي منك فردا في الحجى

من البحر الكامل

إنِّي أحاجي منك فرداً في الحجى … تزهو به الدنيا ونيسابور
فأقولُ ما شجرٌ بديعٌ وصفُهُ … لكنَّهُ ببلادِهِ مقصورُ
أوراقُهُ ورق وعودٌ عودُهُ … والنَّورُ تبرٌ قد علاهُ نورُ
ونشارُهُ الياقوتُ فيه الجزع مَطْ … ويٌّ وبينَهما يُرى الكافورُ

## انظروا كيف تخمد الأنوار

من البحر الخفيف

انظروا كيف تخمد الأنوار … انظروا كيف تسقط الأقمار

## دعوت بماء في إناء فجاءني

من البحر الطويل

دعوتُ بماءٍ في إناءٍ فجاءَني ال … حبيبُ بهِ خَمْراً فأوْسَعْتُهُ زَجْرا
فقالَ هو الماءُ القراحُ وإنَّما … تجلَّى لهُ وجهي فأوهمَكَ الخَمْرا

## فديت مسافرا ركب الفيافي

من البحر الوافر

فديتُ مسافِراً ركِبَ الفيافي … وأَثَّرَ في محاسِنِهِ السِّفارُ
فمسَّكَ وردَ خدَّيهِ السَّوافي … وعَنْبَر مِسْكَ صدغَيْهِ الغبارُ

## يا ليلة كالمسك منظرها

من أحذ

يا ليلةً كالمسكِ منظرُها … وكذاكَ في التشبيهِ مخبرُها
أحييتُها والبدرُ يخدِمُني … والشمسُ أنهاها وآمُرُها

## أظن الربيع العام قد جاء تاجرا

من البحر الطويل

أظنُّ الربيعَ العامَ قد جاءَ تاجِراً … ففي الشمسِ بزَّازاً وفي الريحِ عَطَّارا
وما العيشُ إلاّ أن تواجِهَ وجهَهُ … وتقضي بين الوشي والمسك أوطارا

## قل للمليك الأجل قدرا

من مخلع

قُلْ للمليكِ الأجَلِّ قَدْرا … لا زِلْتَ بَدْراً تحلُّ صَدْرا
إنِّي أُعزِّيكَ عن عزيزٍ … كانَ لريبِ الزمانِ عُذْرا
وكانَ طُهْراً فصارَ أَجْرا … وكانَ ظَهْراً فصارَ ذُخْرا

## سقى الله أياما أشبه حسنها

من البحر الطويل

سقى الله أياماً أُشبِّهُ حُسنها … وقد كنتُ في روض من العيش ناضرِ
بشعرِ ابن معتزٍّ وخطِّ ابن مقلةٍ … ودولةِ مسعودٍ وخلقِ مسافرِ

## فديت غزالا راقني در شعره

من البحر الطويل

فديتُ غزالاً راقني دُرُّ شعرِهِ … كما شاقَني في نطقِهِ دُرُّ ثغرِهِ
إذا ما غدا للشعرِ يُغري بنظمِهِ … غدوتُ لعقدِ الدمعِ أغري بنثرِهِ
وواللهِ ما أدري أسِحْرُ جفونِهِ … تملَّكَ قلبَ الصبِّ أم سحرُ شعرِهِ

## وصولجان بيدي شادن

من البحر السريع

وصولجانٍ بيدَيْ شادنٍ … لا يحسنُ العاشقُ أن يذكرَهْ

## إني أرى ألفاظك الغرا

من البحر السريع

إنِّي أرى ألفاظَكَ الغرَّا … عطَّلَتِ الياقوت والدُّرَّا
لكَ الكلامُ الحرُّ يا من غدا … معروفُهُ يستعبِدُ الحُرَّا

## ودونك بيتا قد تحلت به النهى

من البحر الطويل

ودونكَ بيتاً قد تحلَّت به النُّهى … كما يتحلَّى معصم بسوارِهِ
إذا لم يكنْ في منزلِ المرءِ حُرَّةٌ … تدبِّرُهُ ضاعت مصالحُ دارِهِ

## سماء كصدر الباز والأرض تحته

من البحر الطويل

سماءٌ كصدرِ البازِ والأرضُ تحتَهُ … كأجنحةِ الطاووسِ فاشربْ أبا نصرِ
عقاراً كعينِ الديكِ تحلُو بمسمَعٍ … يؤدِّي غناءَ العندليبِ على قَدْرِ

## خليلي إني من محبتي العلى

من البحر الطويل

خليليَّ إنِّي من محبَّتي العلى … بُليتُ بعلوي الصفات أخي البدرِ
فعقد الثريَّا مستكنٌّ بثغره … ومنطقهُ الجوزاء في خصره تجري
ووجدي به وجد المكارم والعلى … بسيِّدنا الشيخ العميد أبي نصرِ

## أيا يا من مجده للدهر غره

من البحر الوافر

أيا يا مَنْ مجدُهُ للدهرِ غُرَّه … وطلعتُهُ لعينِ المُلْكِ قُرَّهْ
وخدمتُهُ لنارِ العِزِّ زَنْدٌ … وحضرتُهُ لشخصِ المجدِ سُرَّهْ
ويا من ذكرُهُ مثلُ اسمِهِ لا … يزالُ مسافِراً في خيرِ سَفْرَهْ
حويتَ محاسنَ الدُّنيا كما قَدْ … سَبَكْتَ محاسنَ الآداب نُقْرَهْ
وحُزْتَ خصائصَ الرؤساءِ طُرًّا … وحَصَّلْتَ السعودَ لديكَ صَبْرَهْ
ولما لم يسَعْكَ الدهرُ ثَوْباً … قطعتَ لشخصِ مجدِكَ منهُ صُدْرَهْ
وكم لكَ عند عبدِكَ من صنيعٍ … رفيعٍ لا يؤدِّي العبدُ شُكْرَهْ
وذَنْبُ الدهرِ جَلَّ فإن أراني … محيَّاهُ الجميلَ قبلتُ عُذْرَهْ
ظفِرتَ بما تشاءُ من الأماني … فأغمدَ عنكَ صَرْفُ الدهر ظُفْرَهْ
لرأسِكَ خضرةٌ في كلِّ يومٍ … وللكاساتِ فوقَ يديكَ حُمْرَهْ

## إذا الصبا بنسيم الورد والزهر

من البحر البسيط

إذا الصَّبا بنسيمِ الوردِ والزَّهرِ … تعطَّرَتْ وجرَتْ في آخرِ السَّحَرِ
حمَّلتُها من سَلامي ما تبلِّغُهُ … مَن دارُهُ بين سمعِ الملكِ والبصرِ
أعني به الشيخَ مولانا العميدَ أبا … نصرِ بنِ مشكانَ فردَ الدهرِ والبشرِ
إذا تناوبت الأخبارُ مفخرةً … زادتْ محاسنُ لقياهُ على الخبرِ
وإن تداولتِ الأسحارُ سؤددهُ … صلَّى عليهِ لسان الشعرِ في السَّمَرِ

## كتبت إليك من سكر السرور

من البحر الوافر

كتبتُ إليكَ من سكرِ السرورِ … وكاساتٌ تدورُ على بدورِ
وماءُ الوردِ يهطلُ من سحاب ال … بُخورِ على السَّوالفِ والنُّحورِ
وعينُ الدهرِ قد نامتْ فقامتْ … لنا سوقُ الملاهي والسرورِ
وقد قادَ الغلامُ إليكَ طِرفي … فرأيُكَ لا عدمتكَ في الحضورِ

## الغيم بين ممسك ومعصفر

من البحر الكامل

الغيمُ بينَ مُمَسَّكٍ ومُعَصْفَرِ … والماءُ بينَ مُصَنْدَلٍ ومُعَنْبَرِ
والروضُ بينَ مُدَمْلَجٍ ومُتوَّجِ … والوردُ بينَ مدرهَمٍ ومدنَّرِ
والأرضُ قَدْ لَبِسَتْ قميصاً أخضراً … تختالُ فيه بطيلسانٍ أحمرِ
لتروقَنا بظرائفٍ ولطائفٍ … من حسنِ منظرِها وطيبِ المخبرِ
سبحانَ مُحْيِي الأرض بعد مماتِها … وكذاكَ يحيي الخلقَ يومَ المحشرِ

## أنظر إلى البدر في أسر الكسوف بدا

من البحر البسيط

أنظر إلى البدرِ في أسرِ الكسوف بدا … مستسلِماً لقضاء اللهِ والقَدَرِ
كأنَّهُ وجهُ معشوقٍ أدلَّ على … عشاقِهِ فابتلاهُ اللهُ بالشَّعَرِ

## نظرت فلم أجد لك من نظير

من البحر الوافر

نظرتُ فلم أجِدْ لكَ من نظيرِ … ولم أسْمَعْ بمثلِكَ من وزيرِ
كريمِ الخيمِ مرموقِ السجايا … شريفِ المُنْتَمى عفِّ الضميرِ

على الأعداءِ كالقدرِ المُبيرِ … وللأصحابِ كالقمرِ المنيرِ

## كم في ضمير الغيب من أسرار

من البحر الكامل

فاسْتَشْعِرِ الظنَّ الجميلَ تَوَقُّعاً … لمناجحِ الأوطارِ في الأطوارِ

## يا لابسا لنقاب ورد أحمر

من البحر الكامل

يا لابساً لنقابِ وردٍ أحمرِ … يا فارشاً وجهي بوردٍ أصفرِ
حتَّامَ تُنْحِلُني بخصرٍ ناحلٍ … وتُعِلُّني بعليلِ طرفٍ أحورِ
يا واحداً في الحُسْنِ ها أنا واحدٌ … في الحُزْنِ أُصْلى حَرَّ نارٍ مضمرِ
وأَظَلُّ بينَ تذلُّلٍ وتَحَيُّرِ … إذ أنتَ بينَ تدلُّلٍ وتَجَبُّرِ
ما لي بوصفِكَ سيدي من طاقةٍ … ولوَ انَّني استمليتُ طبعَ البُحْتري

## إني بليت بسيد كالدهر إذ

من البحر الكامل

إنِّي بُليتُ بسيِّدٍ كالدهرِ إذْ … يُنْحي بسطوتِهِ على الأحرارِ
فرطُ الفظاظَةِ والصلابةِ دأْبُهُ … وأنا لديهِ بذُلَّةٍ وصَغَارِ

## بي فاقة غطيتها بتجمل

من البحر الكامل

بي فاقةٌ غطَّيْتُها بتجمُّلٍ … وتحمُّلٍ وتَخَمُّلٍ وتَسَتُّرِ
فالحالُ ظاهرُها مروءة موسِرٍ … لكنَّ باطِنَها خَصَاصَةُ ومعسرِ

## أنسيم الرياض حول الغدير

من البحر الخفيف

أنسيمُ الرياضِ حولَ الغديرِ … مازَجَتْ ريَّا الحبيبِ الأثيرِ
أو ورودُ البشيرِ بالنُّجحِ مِن فَ … كِّ أسيرِ أو يسرُ أمرٍ عسيرِ
في ملاءٍ من الشبابِ جديدٍ … تحتَ أيكٍ من التصابي نضيرِ
أم كتابُ الأميرِ سيِّدِنا الغم … رِ فيا حبَّذا كتابُ الأميرِ
وثمارُ الصدورِ وما أجتنيهِ … من سطورٍ فيها شفاءُ الصدورِ
نَمَّقَتْها أناملٌ تفتقُ الأن … وارَ والزهرَ في رياضِ السطورِ
كالمُنى قد جُمعنَ في النِّعَمِ الغُ … رِّ معَ الأمنِ من صروفِ الدهورِ
يا أبا الفضلِ وابنَهُ وأخاهُ … جَلَّ باريكَ من لطِيفٍ خبيرِ
شِيَمٌ يرتضِعْنَ دُرَّ المعالي … ويعبِّرْنَ عن نسيمِ العبيرِ
وسجايا كأنَّهُنَّ لدى النش … رِ رضابُ الحيا بأريٍ مشورِ
ومحيًّا لدى الملوك محيَّا … صادقُ البشر مخجلٌ للبدورِ

## يا رب أنت وهبتها لي نعمة

من البحر الكامل

يا ربِّ أنتَ وهبتَها ليَ نعمةً … أضحَتْ تُعينُ على الزمانِ بِبِرِّها
ووهبتَ منها نعمةً لا تُلْهِني … يا ربِّ أنتَ بسُكْرِها عن شُكرِها

## ويوم سعد حسن البشر

من البحر السريع

ويومِ سَعْدٍ حَسَنِ البِشْرِ … عَذْبِ السَّجايا طَيِّبَ النَّشْرِ
لم تَقْذَ عيني بأذاهُ ولم … يَطِرْ فُؤادي بيَدِ الذُّعْرِ
ولم يَرُعْني لا ولا ساءَني … كعادةِ الأيامِ في الشرِّ
شبَّهْتُهُ منتزعاً من يد ال … أحداثِ ذات الشرِّ والضرِّ
باللبنِ السائغ ذاكَ الذي … من بينِ فرثٍ ودمٍ يجري

## نسيم الصبا وبشير المطر

من البحر المتقارب

نسيمُ الصَّبا وبشيرُ المطرْ … ورَيَّا الرياحينِ حينَ السَّحَرْ
وسجعُ الحمائِمُ فوق الغصون … يجاوبهنَّ لسانُ الوتَرْ
ولفظُ العميدِ ابنِ مشكانَ في … ترسُّلِهِ مرسلاً للغُرَرْ
يذكِّرُني طيب عيشٍ صفا … من الهمِّ قبلَ اعتراضِ الكَدَرْ

## كتبت من صومعة

من مجزوء

كتبتُ من صومعةٍ … تسمحُ بالقوت العَسِرْ
والدهرُ من جفائِهِ … يلبس لي جلدَ النَّمِرْ
فماءُ عيشي كَدِرٌ … ونجمُ حالي مُنْكدِرْ

## بعثت إلى سيدي سكرا

من البحر المتقارب

بعثتُ إلى سيِّدي سُكَّراً … حلاوتُهُ في قرارِ الصدورْ
وشمعاً يمزِّقُ ثوبَ الدُّجى … ويُلْبِسُ جيرانَهُ ثوبَ نورْ

## لك الدنيا وما فيها بلاد

من البحر الوافر

لك الدنيا وما فيها بلاد … تلاحظها بعينيك احتقارا
تكبر ذا الزمان على بنيه … فعش حتى تعلمه الصغارا
وصار صغارهم فيه كبارا … فدم حتى تردهم صغارا
خدمت لك الملوك اروض نفسي … لآمن تحت خدمتك العثارا
ولو كانت لنا الدنيا جعلنا … لك الدنيا وما فيها نثارا

## اخنى عليه الشهر والدهر

من أحذ

اخنى عليه الشهر والدهر … ومحسا محاسن وجهه الشعر
من يصف ما قد دهاه يقل … لا تعجبوا قد يكسف البدر

## بأبي من اذا أراد سراري

من البحر الخفيف

بأبي من اذا أراد سراري … عبرت لي أنفاسه عن عبير
وسباني ثغر كدرٍّ نظيم … تحته منطق كدرٍّ نثير
وله طلعة كنيل الأماني … أو كشعر المهلبي الوزير

## قلت لما أدنت الدنيا لنا

من البحر المتقارب

قلت لما أدنت الدنيا لنا … نفراً ذقنا بهم حرَّ سقر
فاتنا عِزُّ نواصي الخيل فل … يبق فينا ذل أذناب البقر

## لنا ملك تاجه المشتري

من البحر المتقارب

لنا مَلِكٌ تاجُهُ المُشتري … فما أحدٌ غيرُهُ لابِسُهْ
ومُلْكُ الورى فَرَسٌ مُلْجَمٌ … وما أحدٌ غيرُهُ فارِسُهْ
وقد فتحَ الريَّ فراشُهُ … وكرمانُ يَفْتَحها سائِسُهْ

## لك صدغ كأنه قلب فرعون

من البحر الخفيف

لكَ صدغٌ كأنَّهُ قلبُ فرعو … نَ ووجهٌ كأنَّهُ يَدُ موسى
وفمٌ قد أتى ببرهانِ عيسى … فهو بالطيبِ منهُ يُحيي النُّفُوسا

## ألا إن معنى الليث والغيث والشمس

من البحر الطويل

ألا إنَّ معنى الليث والغيث والشمس … بخوارِزْمِشاهٍ غُرَّةِ الجنِّ والإنسِ
ومن عجبي أنِّي إذا ما مدحتُهُ … تشاغلتُ بالتسبيحِ في مجلسِ الأُنسِ

## من رأى غرة العميد أبي نصر

من البحر الخفيف

من رأى غُرَّةَ العميدِ أبي نص … رٍ رأى المشتري ببرج القوسِ
من يُعِرْ كتبةَ مشكان لحظاً … يطَّلِعْ في نموذجِ الفردوسِ
عين ربِّي عليهِ من بدرِ صدرٍ … ودُّه خزرجيٌّ ولقياهُ أوسي
ليس لي طاقة بوصف معالي … هِ ولو كنتُ مفلقاً كابنِ أوسِ

## طالع يومي غير منحوس

من البحر السريع

طالعُ يومي غيرُ منحوسٍ … فاسقني يا طاردَ البوسِ
كأساً كعينِ الديكِ في روضةٍ … كأنَّها حلةُ طاووسِ

## أقول وقد ضاقت باحزانها نفسي

من البحر الطويل

أقول وقد ضاقت باحزانها نفسي … لئن بعت يا مولاي ودي بالوكس
لقد بيع بعض الأنبياء عليهم … صلاة اله الناس بالثمن البخس

## قل للذي أنا في هواه خاشي

من البحر الكامل

قُلْ للذي أنا في هواهُ خاشِي … صادَ الفؤادَ بصدغِهِ الجَمَّاشِ
قلبٌ يُرى عند الرياحِ كأنَّه … قلبُ ابنِ سيمجور أحس بتاشِ

## إن الشتاء مضى بقبح فاشي

من البحر الكامل

إنَّ الشتاءَ مضى بقبحٍ فاشي … وأتى الربيعُ لنا بحسنِ رياشِ
ومضى ابنُ سيمجور بقبحِ فعالِهِ … وانتاشَ أبناءُ الكرامِ بتاشِ

## صباح محاسنه تستفيض

من البحر المتقارب

صباحٌ محاسِنُهُ تستفيض … وروضٌ أريضٌ وغيمٌ يفيض
فكيفَ الوفاءُ بما يقتضيهِ … وحالَ الجريضُ دوينَ القريضْ
فأُنسي مريضٌ وهمِّي عريض … وطرفي غضيضٌ وعظمي مَهيض

## فضفضت ختام القلب مني وحزته

من البحر الطويل

فضفضتَ خِتامَ القلبِ منِّي وحُزْتَهُ … جميعاً ولا واللهِ غيرُكَ ما فضَّهْ
ولما نثرتَ المِسْكَ من فوقِ فِضَّةٍ … نَثَرْتَ على مِسْكي نثاراً من الفضَّهْ

## جالسني شادن كلفت به

من البحر المنسرح

جالسني شادنٌ كَلِفْتُ بهِ … في صفةٍ حالُنا بها غَضَّهْ
دمعيَ ياقوتةٌ على ذهبٍ … وفوهُ ياقوتةٌ على فِضَّهْ

## أيا قاضيا قد دنت كتبه

من البحر المتقارب

أيا قاضياً قد دنت كُتُبه … وان اصبحت داره شاحطه
كتاب الوساطة في حسنه … لعقد معاليك كالواسطه

## وليل كلون الظبي غيرت لونه

من البحر الطويل

وليلٍ كلونِ الظَّبْيِ غَيَّرْتُ لونَهُ … براحٍ كعينِ الديكِ بل هُوَ ألمَعُ
ولمَّا مَزَجْتُ الرُّوحَ منِّي براحِها … تَرَحَّلَ عنِّي الهمُّ والغَمُّ وأجْمَعُ

## يا بدر صدر بنيسابور مطلعه

من البحر البسيط

يا بدرَ صَدْرٍ بنيسابورَ مطلعُهُ … وبحرَ جودٍ لأهلِ الفضلِ مترعُهُ
سقيت كرميَ ماءً فيه أربعةٌ … منَ المياهِ وخيرُ الماء أنفعُهُ
ماءُ الحياةِ وماءُ الوجهِ يشفعُهُ … ماءُ الشبابِ وماءُ الوردِ يتبعُهُ
بقيتَ ما بقيَتْ نفسٌ وما طَلَعَتْ … شمسٌ وما سارَ من مدحيك أبدعُهُ
للعرفِ تصنعُهُ والخير تزرعُهُ … والمجدِ تجمعُهُ والمدحِ تسمعُهُ

## وموقف للوداع ألبسني

من البحر المنسرح

وموقفٍ للوداعِ ألبَسَني … للتُّعْسِ همًّا يسوءُ موقعُهُ
فقلتُ والدمعُ قد شَرِمْتُ به … أستودعُ اللهَ من أُودِّعُهُ

## يا من كساه الله أردية العلى

من البحر الكامل

يا منْ كساهُ اللهُ أرديةَ العُلى … وحباهُ عطرَ ثنائِها المتضوعِ
وإذا نظرتُ إلى محاسنِ وجهه ال … مسعود قلت لمقلتي فيها ارتعي
وإذا قريتُ الأذنَ شهدَ كلامِهِ … قلتُ اسمعي وتمتَّعي وارعَيْ وَعِي
فكأنَّما يُوحي إلى خَطَراتِهِ … في مطلعٍ أو مَخْلَصٍ أو مقطعِ
لكَ في المحاسنِ معجزاتٌ جَمَّةٌ … أبداً لغيركَ في الورى تُجْمَعُ
بحران بحرٌ في البلاغةِ شابَهُ … شعر الوليد وحسن لفظ الأصمعي
وترسُّل الصابي يزينُ علوَّه … خطُّ ابنِ مقلةَ ذي المحلِّ الأرفعِ
كالنَّوْرِ أو كالسِّحرِ أو كالبدرِ أو … كالوشيِ في بردٍ عليهِ موشَّعِ
شكراً فكمْ من فِقْرةٍ لكَ كالغِنى … وافى الكريمَ بعيدَ فقرٍ مدقعِ
وإذا تفتَّقَ نَورُ شعرِكَ ناضراً … فالحسنُ بينَ مرصَّعٍ ومُصَرَّعِ
أرجلت فرسانَ القريضِ ورضت أف … راسَ البديعِ وأنتَ أفرسُ مبدعِ
ونَقَشْتَ في فصِّ الزمانِ بدائعاً … تُزْري بآثارِ الربيعِ المُمْرَعِ
وحويت ما تُكْنى به طرًّا فلم … تَترُكْ لغيرِك فيهِ بعضَ المطمعِ

## يا مهدي الطرف الجواد كأنما

من البحر الكامل

يا مهديَ الطّرفِ الجوادِ كأنَّما … قد أنْعَلُوهُ بالرياحِ الأربعِ
كالجاحمِ المشبوبِ أو كالهاطل ال … مصبوبِ أو كالباشقِ المتسرعِ
لا شيءَ أسرعُ منه إلاّ خاطري … في شكرِ نائِلكَ الجليلِ الموقعِ
ولوَ انَّني أنصفت في إجلالِهِ … لجلالِ مهديهِ الهمامِ الأروعِ
أقْضَمْتُهُ حَبَّ الفؤاد لحبِّه … وجعلتُ مربطَهُ سوادَ المدمعِ
وخلعتُ ثمَّ قطعتُ غيرَ مضيِّقٍ … بُرْدَ الشباب لجلِّهِ والبرقعِ

## هل سبيل إلى عناق كما عانقت

من البحر الخفيف

هل سبيلٌ إلى عناقٍ كما عا … نقتُ عندَ الفراقِ يومَ الوداعِ
شادناً فاتناً سميناً جسيماً … ملءَ عيني وملءَ قلبي وباعي

## رمضان أرمضني فأمرضني بصادات

من البحر الكامل

رمضانُ أرمَضَني فأمرَضَني بصا … داتٍ على عددِ الطباعِ الأربعَهْ
صومٌ وصفراءٌ تجرِّعُني الردى … وصبابةٌ وصدودُ من قلبي مَعَهْ

## وقالوا افترشت النطع وقد أتى

من البحر الطويل

## عشقت الجود جدا فهو طبعك

من البحر الطويل

## يا من له كل الذي يكنى به

من البحر الكامل

يا مَنْ له كلُّ الذي يكنَى بهِ … ومفرّق الدُّنيا لديهِ مؤلَّفُ

وتصرَّفَتْ بكَ في المكارمِ والعُلى … هِمَمٌ على قِمَمِ النجومِ تَصَرَّفُ
وملكتَ أحرارَ الكلامِ كأنَّها … خدمٌ وغلمانٌ لأمركَ وُقَّفُ
وكأنَّما نَوْرُ الربيعِ وزهرُهُ … من وشيِ خطِّكَ في المهارقِ أحرفُ

## لقد أشربت قلبي الحب صرفا

من البحر الوافر

لقد أشربتَ قلبي الحبَّ صِرفا … بشارِبِكَ الذي يزدادُ ظرفا

## اسمع فديتك حلفة مبرورة

من البحر الكامل

## يا سيدا حاز طبعه الشرفا

## ويوم عبيري النسيم سبى طرفي

من البحر الطويل

لما رآني وفي يدي قلمٌ … لم يَدْرِ مولاي أيُّنا القلمُ

## يا سائلي وصف مولانا أبي الحسن

من البحر البسيط

يا سائلي وصفَ مولانا أبي الحسنِ … مسافرٍ في بديعِ القولِ محكمِهِ
المسكُ من ذِكْرِهِ والمزنُ من يدِهِ … والروضُ من خُلْقِهِ والدرُّ من فمِهِ

## سعدت بغرة وجهك الأيام

من البحر الكامل

سَعِدَتْ بغُرَّةِ وجهِكَ الأيَّامُ … وتزيَّنَتْ ببقائِكَ الأعوامُ
وتصرَّفَتْ بكَ في المعالي هِمَّةً … تعيا بها الأفهامُ والأوهامُ
ولقد فَرَشْتَ مهادَ عدلِكَ فاغْتَدَتْ … تتواردُ الآسادُ والآرامُ
وافتضَّ سيفُ عُلاكَ كلَّ مدينةٍ … بكرٍ عليها للإياسِ خِتامُ
هذي زرنجُ استغلقت وتمنَّعَتْ … فكأنَّها إلاّ عليكَ حرامُ
ففتحتَها وأَبحتَها ومَنَحتَها … نفراً هُمُ لفنائِكَ الخدَّامُ
وقَدِمتَ والأيامُ تُنْشِدُ في الورى … بيتاً تجيدُ نشيدَهُ الأيامُ
قد جاءَ نصرُ اللهِ والفتحُ الذي … تُزْهى بكتبةِ وصفِهِ الأقلامُ
بأجَلِّ أحوالٍ وأيمنِ مقدمٍ … وأتمِّ إقبالٍ يليه دوامُ

## وسكباجة تشفي السقام بطيبها

من البحر الطويل

وسكباجةٍ تشفي السَّقَامَ بطيبِها … على أنَّها جاءَتْ بلونِ سقيمِ
إذا زارها أيدي الرجالِ تزاحَمَتْ … كأيدي نساءٍ في ظلالِ نعيمِ

## وحمام له حر الجحيم

من البحر الوافر

وحمَّامٌ له حرُّ الجحيمِ … ولكنْ شابَهُ بردُ النعيمِ
رأيتُ به ثواباً في عقابٍ … وزُرْتُ به نعيماً في جحيمِ

## عركتني الأيام عرك الأديم

من البحر الخفيف

عركتني الأيامُ عَرْكَ الأديمِ … وتجاوَزْنَ بي مدى التقويم
وغضَضْنَ اللحاظَ منِّيَ إلاّ … عن هلال يرنو بمقلةِ ريمِ
لحظُهُ سُقْمُ كلِّ قلبٍ صحيحٍ … ثغرُهُ بردُ كلِّ جسمٍ سقيمِ

## يا من تشابهت المحاسن والعلى

من البحر الكامل

يا من تشابهت المحاسنُ والعُلى … فيهِ وأصبحتِ القلوبُ برسمِهِ
فالخُلقُ من كخَلْقِهِ والخَلْقُ مِنْ … هُ كلفظِهِ والشِّعْرُ منه كاسمِهِ
وغذاءُ جسمي من سماحِ يمينِهِ … وغذاءُ روحي من بدائعِ نظمِهِ
لا زِلْتَ بينَ سعادةٍ وزيادةٍ … وسلِمْتَ من سيفِ الزمانِ وسهمِهِ

## فديوان الضياع بفتح ضاد

من البحر الكامل

فديوانُ الضِّياعِ بفتحِ ضادٍ … وديوانُ الخِراج بحذفِ جيمِ

## رب يوم هواؤه يتلظى

من البحر الخفيف

ربَّ يومٍ هواؤُه يتلظَّى … فيحاكي فؤادَ صَبٍّ متيَّمِ
قلتُ إذا صَكَّ حَرُّه حُرَّ وَجْهي … ربَّنا اصرِفْ عنَّا عذاب جَهَنَّمْ

## قد سقتنا السماء ماء الغيوم

من البحر الخفيف

قد سقتنا السماءُ ماءَ الغيومِ … فاسقِنا يا غلامُ ماءَ الكرومِ
نشربُ الراح بادِّكار الرئيس ال … فَرْدِ في الجودِ والعُلى والعلومِ
وإذا ما مسافرٌ سافَرَتْ أخْ … بارُ علياهُ أخبرَتْ عن نجومِ

## هات في غرة المحرم

من مجزوء

هات في غُرَّةِ المحرَّ … مِ عينَ المحرَّمِ
واسقني الكأسَ قد أُشَبِّ … هها في توهُّمي
بنسيمٍ معقَّدٍ … في هواءٍ مُجَسَّمِ

## رغيفك في الأمن يا سيدي

من البحر المتقارب

رغيفك في الأمن يا سيدي … يحل محل قمام الحرم
فلله دَرُّك من سيدٍ … حرام الرَّغيف حلال الحُرم

## سأرسل بيتا يجمع الصدق والحسنا

من البحر الطويل

سأرسل بيتاً يجمعُ الصدقَ والحُسنا … على لوعةٍ تستغرِقُ اللبَّ والذهنا
غدوتُ نحولاً واصفراراً كتبنةٍ … وفوكَ بجاديُّ غدا يجذِبُ التبنا

## لقد قال دهقان لكسرى بن هرمز

من البحر الطويل

لقد قالَ دهقانٌ لكسرى بنِ هرمزٍ … ونكَّتَ في التجويدِ فيه وأحْسَنا
لقد غَرَسُوا حتَّى أكلنا وإنَّنا … لَنَغْرُسُ حتَّى يأكُلَ الناسُ بعدَنا

## وتبرية الرأس فضية العجيزة

من البحر المتقارب

وتبريةِ الرأس فضية ال … عجيزةِ فيروزجٌ عينُها
إذا طَلَعَتْ سرَّني قُرْبُها … وإنْ غربتْ ساءني بينُها

## أبا سليمان سر في الأرض فأقم

من البحر البسيط

أبا سليمانَ سِرْ في الأرضِ فأَقِمْ … فأنتَ عندي دَنَا مثواكَ أو شطَنَا
ما أنتَ غيري فأخشى أن أُفارِقَهُ … فديتُ روحَكَ بل رُوحي فأنتَ أنا

## وند ما له ند

من البحر الهزج

ونَدٍّ ما له نِدُّ … تعاطيهِ من السُّنَّه
إذا ما جاورَ النار … حكى رائحةَ الجنَّه

## بدر خلعت على الزمان رداءه

من البحر الكامل

بدرٌ خلعتُ على الزمانِ رداءهُ … فسرى وسارَ بألسنِ الكتَّانِ
صدرُ الوزارةِ قَدْ بدا في سعدِهِ ال … سَعْدانِ والقمران والعمرانِ

## لا كان في عيني مجال للسنه

من البحر الكامل

لا كانَ في عيني مجالٌ للسِّنَه … وجعلتُ عِرضي طُعْمَةً للألسِنَه
إن ذقت طعمَ العيشِ بعدكَ ساعةً … ورأيتُ يومَ البينِ إلاّ كالسَّنه

## باح الصباح بأسرار البساتين

من البحر البسيط

باحَ الصباحُ بأسرارِ البساتينِ … وأحيتِ النفسُ أنفاسَ الرياحينِ
حتَّى حسبتُ نسيمَ الروضِ أقرأَني … كتبَ ابنِ مشكانَ عن صدرِ السلاطينِ
فهاتِها كشعاعِ الشمس صافيةً … بكراً ربيبةً أبيات الدهاقينِ
نشربْ على ذكر عنوان المحاسن قا … نونِ الفضائلِ منظورِ الدواوينِ

## أما ترى الشمس حلت الميزانا

من البحر الخفيف

أما ترى الشمسَ حلَّتِ الميزانا … في زمانٍ قد عدَّل الميزانا
ومن العدلِ شربُك الراح في وق … تِ اعتدالٍ ونَفْيُكَ الأحزانا
إنَّ هذا الخريفَ وهو ربيع ال … ناس يدعو الأرطال والكيزانا

## هي القناعة فالزمها تعش ملكا

من البحر البسيط

هي القناعةُ فالزَمْها تعِشْ ملكاً … لو لم يكنْ منكَ إلاّ راحةُ البدنِ
وانظُرْ إلى مالكِ الدنيا بأجْمعِها … هل راحَ منها بغيرِ القُطْنِ والكفنِ

## من مبلغ الصدر مولانا أبا الحسن

من البحر البسيط

من مبلغُ الصدرَ مولانا أبا الحَسَنِ … مسافِراً نُكْثَةُ الأيَّامِ والزمنِ
خفَّفْتُ ظهريَ من ثِقْلِ الكروبِ كما … أثْقَلْتَهُ بالأيادي الغُرِّ والمننِ
صنائعٌ منكَ جَلَّتْ في الأنامِ وقَدْ … دقَّتْ معانيكَ في الأشعارِ والفطنِ
وقد أتاني قريضٌ قد نَفَثْتَ بهِ … كالسحرِ والراحِ والريحانِ في قَرَنِ
واللهُ يجزيكَ عن عبدٍ ومُصْطَنَعٍ … قد كانَ مَيْتاً بأيدي البثِّ والحَزَنِ
فعاشَ من كلماتٍ منكَ كُنَّ له … كالروحِ عائدةً منهُ إلى بَدَنِ

## هذا غزال الهند في الغزلان

من البحر الرجز

هذا غزالُ الهندِ في الغزلانِ … كُمِثْلِ عُودِ الهندِ في العيدانِ
وَجْهٌ بديعُ الحُسْنِ في الغلمانِ … مصوَّرٌ من حَدَقِ الحسانِ
مركَّبٌ في مُلَحِ الخيلانِ … كأنَّه في ناظر الإنسانِ
إنسانُ عينِ الحُسْنِ في الزمانِ

## علي بند كصفو الزمان

من البحر المتقارب

عليَّ بندٍّ كَصَفوِ الزمانِ … ونيل الأماني وحِرْزِ الأمان
إذا نالتِ النارُ من جسمِهِ … أَتَتْ ريحُهُ بنسيمِ الجنانِ

## عندي إنسان ولكنه

من البحر السريع

عنديَ إنسانٌ ولكنَّه … أكثرُ لي من ألفِ إنسانِ
لقاؤهُ أشهى من البارد ال … عَذْبِ إلى عطشانِ غصانِ
فاقتَرِنا عندي أفدِّيكما … فأَنتُما راحِي ورَيْحاني

## لي صاحب لا يسمى

من البحر المجتث

لي صاحب لا يُسمَّى … بين الورى إنسانا
لأنه التَّيسُ قرناً … ولحيةً وصناناً

## يا قبلة العشاق يا من به

من البحر السريع

يا قِبْلَةَ العشَّاقِ يا مَنْ بهِ … سِتْرُ الهوى بينَ الوَرَى مُنْتَهكْ
جَرَّدْتَ من عَيْنَيْكَ سَيْفاً فَلِمْ … أغْمَدْتَهُ في قلبِ عبدِ الملِكْ

## وليلة ذات أضواء وضوضاء

من البحر البسيط

وليلةٍ ذاتِ أضواءٍ وضوضاءِ … تجرُّ أذيالَ لألاءِ وآلاءِ
تصالحَ الليلُ فيها والنهارُ على … تأليفِ ما بينَ إصباحٍ وإمساءِ
كأَنَّما الشمسُ سكرى فهي راجعةٌ … من قبلِ موعدِها تومي بأضواءِ
كأنَّما الأرضُ شحراء من الذهب ال … إبريز سامية نحو الغميضاءِ

## أيا طيب عيشي أرى بركة

من البحر المتقارب

أيا طيبَ عيشي أرى بركةً … تسوق إلى روضِها ماءها
إذا أنت واجهتها في الدُّجى … حسبتَ الكواكبَ حصباءها

## يا ليلة قدمت حتى إذا قدمت

من البحر البسيط

يا ليلةً قُدِّمت حتَّى إذا قَدِمت … عظّمتُ حرمتَها فِعْلَ الألِبَّاءِ
لأنَّها بادّكارٍ للأوائلِ من … صِيْدِ الملوكِ الأشدَّاءِ الأسدّاءِ
قد أطلعتْ من عِنان الراح طلعتَها … وأَنطقتْ أَلْسُنَ العيدانِ والناءِ
ونبَّهتْ أعينَ اللذات من سِنَةٍ … والشأنُ في نظمِها عقدُ الأحِبَّاءِ

## يا عمدة الأمراء والوزراء

من البحر الكامل

يا عمدةَ الأمراءِ والوزراءِ … يا عِدَّةَ الأدباءِ والشعراءِ
يا غرَّةَ الزمنِ البهيمِ وناظرَ ال … كرمِ الصميمِ وواحدَ الفضلاءِ
أرأيتَ همَّةَ عقربٍ دَبَّتْ على … قدمٍ بها تخطو إلى العلياءِ
لما ارْتَقَتْ باللَّسْعِ أعظمَ مُرْتَقى … أخْنَتْ عليها رُتْبَةُ العُظَماءِ
إنْ ذُقْتَ ضَرَّاءَ العقاربِ فابْقَيَنْ … بعقاربِ الأصداغِ في السَّرَّاءِ
يا طيبَ لسعةِ عقربٍ ترياقها … ريق الحبيبِ بقهوةٍ عذراءِ

## يا دهر ويحك قد أطلت جفائي

من البحر الكامل

يا دهرُ ويحَكَ قد أطَلْتَ جَفائِي … وتَرَكْتَ ماءَ معيشَتي كجُفاءِ
أتراكُ تحسِبُ أنَّني من جُمْلةِ ال … كُتَّابِ والأدباءِ والشعراءِ
حتى تعادِيَني كعادتِكَ التي … أَنْحَتْ عوادِيها على الفُضَلاءِ
هيهاتَ قد أحْسَنتني ما كنتُ أُح … سِنُهُ فرِفْقاً لَسْتُ في الأُدباءِ

## أقول إذ سألوني عن مروءة من

من البحر البسيط

أقول إذ سألوني عن مروءةِ من … ما لا يقاسُ بأندادٍ وأكفاءِ
محمدٌ لمروءات الأنام غدا … كالزَّنْدِ للنارِ والينبوعِ للماءِ

## قولا لشاعرنا الثقيل الاول

من البحر الكامل

قولا لشاعرنا الثقيل الاول ال … مربى بطلعته على الرقباء

## عذيري من الأيام مدت صروفها

من البحر الطويل

عذيري من الأيَّامِ مَدَّتْ صروفُها … إلى وجهِ من أهوى يَدَ النسخِ والمَحْوِ

فذاكَ سوادُ الحظِّ ينهى عن الهوى … وهذا بياضُ الوَخْطِ يأمرُ بالصحوِ

## إليك المشتكى لا منك ربي

من البحر الوافر

إليكَ المشتكى لا منكَ ربِّي … وأنتَ لحادِثاتِ الدهرِ حسبي
تُروِّي غلَّتي وتَرمُّ حالي … وتُؤْمِنُ رَوْعَتي وتُزِيلُ كَرْبي

## يا رقعة طويت على حيات

من البحر الكامل

يا رقعةً طُوِيَتْ على حَيَّاتِ … وعقاربٍ كَرَّرْنَ ماءَ حياتي
ما أنتِ إلاّ من تباريحِ الجوى … وسَفَاتجِ الأحزان والحَسَراتِ
وكأنَّ أحرفَكِ الكريهةَ أعينٌ … لرواقبٍ أو ألسُنٌ لوُشاةِ
وكذا الضياعُ رقاعُ قيمتِها إذا … وافَتْ أتَتْ بحوادِثِ الآفاتِ

## ألا عين الإله على همام

من البحر الوافر

ألا عينُ الإله على هُمامٍ … إليه في العلى والمجد نُوحي
ومن أضيافِهِ الأشرافِ منهُمْ … سليمانُ بنُ منصورِ بنِ نوحِ

وقال أيضاً: … الوافر
ألا عينُ الإله على هُمامٍ … إليه في العلى والمجد نُوحي

## أنت يا صاح لست عندي بصاح

من البحر الخفيف

أنتَ يا صاحِ لستَ عندي بصاحِ … أنتَ روحي وراحتي أنتَ راحي

## غناؤك غنيتي عن كل زاد

من البحر الوافر

غناؤُك غُنْيَتي عن كلِّ زادِ … ورقصُكَ قد تعلَّمَهُ فؤادي

## يا ليلة هي طولا

من البحر المجتث

يا ليلةً هي طولاً … كمثلِ شوقي وَوَجْدي
مدت سرادق وشيٍ … على الورى أيّ مد

## يا زمانا نعيمه

يا زماناً نعيمُهُ … لم يُعَرِّجْ على يَدي
كنسيمٍ مُعَقَّدٍ … وشعاعٍ مُجَسَّدِ
طيبُهُ كالكَرَى يُلِ … مُّ بجَفْنِ المُسَهَّدِ
أو كخلق المؤمل ب … نِ الخليلِ بن أحمدِ

## قد أقبل الصيف يحكي حر أنفاسي

من البحر البسيط

قد أقبلَ الصيفُ يحكي حَرَّ أنفاسِي … وفي فؤادي حَرٌّ ما لهُ آسي
فإن سمعتَ ببردِ الوصلِ منك فقد … سلَلْتَ نِضوَ رجائي من يَدَيْ ياسي

## ثلاث قد منيت بها فأضحت

من البحر الوافر

ثلاثٌ قد مُنِيتُ بها فأضحت … لنارِ القلبِ منِّي كالأثافي
ديونٌ أنْقَضَتْ ظهري وجَوْرٌ … من الجيران شابَ لهُ غُدافي
وفقدان الكفاف وأيُّ عيشٍ … لمن يُمنى بفقدان الكفافِ

## أقول لدهر وهو يخفض رتبتي

من البحر الطويل

أقولُ لدهرٍ وهو يخفِضُ رُتْبتي … ويُنحي على مالي ويخلِفُ تأميلي
أيا حجراً صلداً مُنِيتُ ببخلِهِ … فلا هو يوريني ولا هو يوري لي

## أبلى جديدي هذان الجديدان

من البحر البسيط

أبلى جديديَّ هذان الجديدانِ … والشأنُ في أن هذا الشيبَ ينعاني
كأنَّما اعتمَّ رأسي منه بالجبلِ ال … راسي فأوهمني ثقلاً وأوهاني

## أقول والقلب مكدود بأحزان

من البحر البسيط

أقول والقلبُ مكدودٌ بأحزان … والصبرُ أبعدُ ممَّا بينَ أجفاني
حتَّى متى أنا يُدمي العضّ أنملَتي … غيظاً على زمنٍ قد رام إزماني
في كلِّ يومٍ أراني في نوائبِهِ … كأنَّني إصبعي والدهر أسناني

## ومهفهف فتن الإله عباده

من البحر الكامل

ومُهَفْهَفٍ فَتَنَ الإِلهُ عبادَهُ … إذ ساقَ حسنَ العالمينَ إليهِ
فكأنَّ بابلَ أصْبَحَتْ في طَرْفِهِ … وكأنَّما الأهوازُ في شفتيهِ
وكأنَّ توقيعَ الرئيسِ مسافرٍ … في عُرْضِ عارضِهِ يلوحُ عليهِ

## إن غبت عنك شكوتني

من مجزوء

إن غِبتُ عنك شكوتني … وإذا وَصَلْتُ هَجَرْتني
وتظلُّ لي مُسْتَبْطِئاً … فإذا حضرتُ حجبتني

## يا من جميع الحسن بعض صفاته

من البحر الكامل

يا من جميعُ الحُسْنِ بعضُ صفاتِهِ … وحلاوةُ الدُّنيا تذاقُ بفيهِ
لا تمرضن جسمي فإنَّكَ روحهُ … لا تُحْرِقَنْ قلبي فإنَّكَ فيه

## سل النجوم التي أراعيها

من البحر المنسرح

سل النجوم التي أراعيها … عن ليلة الهجرِ كيفَ أقضيها
هنَّ شهُودي على سُهودي وال … دِماءُ من مقلتيَّ أُجْرِيها

## ألا يا غرة العليا

من البحر الهزج

ألا يا غُرَّةَ العَليا … ألا يا نُكْتَةَ الدُّنيا
وشمسَ الأرضِ فردَ الدَّه … رِ عينَ السؤدَدِ اليمنى
أما استحيا أبو يحيى … لفَضِّ المهجة الكُبرى
لئِنْ فُتِحت بك الدُّنيا … لقد فُتِحَتْ بكَ الأُخرى

## ديباجة الوجه من علي

من مخلع

ديباجة الوجه من عليٍّ … معمولة في طراز ربِّي
فحُسنه مِلء كل عينٍ … وحبه مِلء كل قلب

## وعجزي بان عن وصل الأيادي

من البحر الوافر

وعجزي بان عن وصل الأيادي … كجار أبي دوادٍ للأيادي

## لنا شيخ بفقحته يواسي

من البحر الوافر

لنا شيخ بفقحته يواسي … ويحلق شاربيه بالمواسي
اذا بايته في جوف بيت … فسا يفسو فساءً فهو فاسي

## عندي إخوان وما بينهم

من البحر السريع

عندي إخوان وما بينهم … إِلا أخ للانس أخِيَّه
وما لجمع الشمل منا سوى … راح صراح في صُراحيَّه

## سئمت العيش حين رأيت

من مجزوء

سئمت العيش حين رأي … ت صرف الدهر يرهقني
صعوداً والصعود إلي … ه يعجزني فيقلقني
وبنتُ الموت بالآلا … م والأوجاع تطرقني
تُؤرقني تحرِّقني … تُعرقني تُغرقني

## إليك قولا سديدا

من البحر المجتث

إليكَ قولاً سَديداً … يروي العِطاشَ بمائِهْ
إن الخِراجَ خُراجٌ … دواؤُه في أدائِهْ
