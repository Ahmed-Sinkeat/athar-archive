## يقول عامر هو الانبوطي

من البحر الرجز

يــقــول عــامــر هـو الانـبـوطـي … أحــمــد ربــي لســت بـالقـنـوطـي
وأســتــعــيـــن الله فــي ألفـيـة … مــقـاصـد الأكـل بـــهـا مـحـويـة
فيـــها صنوف الأكل والمطـاعم … لذت لكـــــــل جـــــــائع وهــــــائم
طـعـامـنـا الضـانـي لذيـذ للنـهمْ … لحـمـا وسـمـنا ثم خبزا فالتقـمْ
فــإنــهــا نـفـيـسـة والأكـل عــمّ … مـطـاعـمـا إلى سـنـاها القلب أمّ
والأصـل فـي الأخـبـاز أن تقمّرا … وجــوزوا التـقـديـد إذ لا ضـررا
وامـنـعـه حـيـن يـسـتـوي الخرفان … فــإنــه يــعــيـق أكـل الضـــانـي

## بشرى الربيع لقد وافت بشائره

من البحر البسيط

بـشــرى الربـيع لقــد وافـت بـشائره … وفـاح دونـــك فـي الآفـاق عــاطــره
ومـالت القـضـب بالأطــيار مـطربـة … وقـد تــبــسم من عجــب أزاهــــــره
فــسر مـقـدمــه الحــالي أخا شــجـن … يـهــيـجـه من مـعاني الدوح نـاضـره
والزهـر من فـرح أهـدى النثـار بـها … لمـا سما الورد واستـعلت مـظـاهـره
حـكـى بـمـنـظــره الحـالي ومخـبـره … صفـات رضوانـنا السامــي زواهره
أمـير مجد لنـا تــتـلى مـــــدائحــــــه … مدى الزمـان كما تـــروى مـــآثـــره
تـخـاله الليــث والمـريـخ فــي يــــده … إذا بدا جـائلاً والســيـف شـاهـــــره
روض نـضـيـر ولكـن مـثـمـر أبـــداً … غـيــث ولكـن نــدى عمــت مواطـره
خــذ مـن زمانك مـا أغـنـاك مغتنـمــا … وأنــت نـاه لهـذا الدهـر آمــــــــــــره
ودم بـروض العـلا والعز مـنـبــسـطاً … بـمطـربـات الهــنـا يــشـدوك طــائره

## أكلت من الضأن رطلين

من البحر عدد

أكلت من الضأن رطلين … يزيد قلبك نفاسه
وابعد عن الكشك يا زين … دا الأكل منه تعاسه

## اجتنب مطعوم عدس وبصل

من البحر الرمل

اجتنب مطعوم عـدس وبصل … فـي عـشـاء فـهو للعقل خبــل
وعن البـيـسـار لا تحفل بــه … تُمس في ضعف وسقم وعلل
واحتفل بالضأن إن كنت فتى … زاكي العقل ودع عنك الكسل
مـن كـباب وضلوع قـد زكـت … أكلها ينفي عـن القلب العلل

## أناجر الضأن ترياق من العلل

من البحر البسيط

أنـاجـر الضـأن تـريـاق مـن العلل … وأصـحـن الرزّ فـيـهـا مـنتهى أملي
أكلي غداءً وأكلي في العشاء على … حـدٍّ سـواء إذا اللحم السمين قلي
فـيـم الإقامة بالأرياف لا شبعي … فـيـهـا ولانـزهتي فيها ولا جذلي
نـاءٍ عـن الأهل خال الجوف منقبض … كــمـعـدم مـات مـن جـوع ومـن قـشـل
فـلا خـليـل بـدفـع الجـوع يرحمني … و لا كـريـم بلحم الضأن يسمح لي
طـال التـلهـف للمـطـعـوم واشتعلت … حُـشـاشـتـي بـحـمام البيت حين قلي
أريـد أكـلا نـفـيـسـا أسـتـعين به … على العبادات والمطلوب من عملي
والدهـر يـفـجـع قـلبـي من مطاعمه … بالعدس والكشك والبيسار والبصل
