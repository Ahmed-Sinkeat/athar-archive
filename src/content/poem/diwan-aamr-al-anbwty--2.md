من البحر البسيط

بـشــرى الربـيع لقــد وافـت بـشائره … وفـاح دونـــك فـي الآفـاق عــاطــره
ومـالت القـضـب بالأطــيار مـطربـة … وقـد تــبــسم من عجــب أزاهــــــره
فــسر مـقـدمــه الحــالي أخا شــجـن … يـهــيـجـه من مـعاني الدوح نـاضـره
والزهـر من فـرح أهـدى النثـار بـها … لمـا سما الورد واستـعلت مـظـاهـره
حـكـى بـمـنـظــره الحـالي ومخـبـره … صفـات رضوانـنا السامــي زواهره
أمـير مجد لنـا تــتـلى مـــــدائحــــــه … مدى الزمـان كما تـــروى مـــآثـــره
تـخـاله الليــث والمـريـخ فــي يــــده … إذا بدا جـائلاً والســيـف شـاهـــــره
روض نـضـيـر ولكـن مـثـمـر أبـــداً … غـيــث ولكـن نــدى عمــت مواطـره
خــذ مـن زمانك مـا أغـنـاك مغتنـمــا … وأنــت نـاه لهـذا الدهـر آمــــــــــــره
ودم بـروض العـلا والعز مـنـبــسـطاً … بـمطـربـات الهــنـا يــشـدوك طــائره
